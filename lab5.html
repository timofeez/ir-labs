ЛР5. Закон Ципфа#

Задание (lesson-04/homework.md): построить график распределения терминов по частотам в логарифмической шкале, наложить закон Ципфа, объяснить причины расхождения. Дополнительно — подобрать константы закона Мандельброта и наложить его на график.

1. Закон Ципфа#

Отсортируем все слова корпуса по убыванию частоты: самое частое получает ранг r = 1, следующее — r = 2 и т. д. Закон Ципфа (1949): частота обратно пропорциональна рангу:

$$ f(r) = \frac{C}{r} \quad\Longleftrightarrow\quad f \cdot r \approx \text{const} $$

Второе по частоте слово встречается примерно вдвое реже первого, десятое — в 10 раз реже. В логарифмических осях это прямая с наклоном −1: $\log f = \log C - \log r$.

Смысл: распределение частот слов — степенной закон, а не нормальное распределение. Немного очень частых слов (служебные) и огромное число редких. Это свойство почти любого естественного текста; Ципф объяснял его «принципом наименьшего усилия».

2. Закон Мандельброта#

Уточнение, которое лучше описывает начало списка (самые частые слова):

$$ f(r) = \frac{P}{(r + \rho)^{B}} $$

ρ сдвигает начало (сглаживает «полку» у служебных слов), B — наклон (у Ципфа B = 1, ρ = 0). В лекции для другого корпуса: B = 1,15, ρ = 100.

3. Подсчёт частот — termfreq.exe (C++, без STL)#

Тот же токенизатор, что в индексаторе; частоты — в свою хеш-таблицу TermTable, затем номера термов сортируются слиянием по убыванию частоты (стабильно: при равной частоте — в порядке первого появления). Выход — freq.tsv: строки «ранг \t частота \t терм».

// Сортировка слиянием номеров термов по убыванию частоты (стабильная: при равной частоте —
// в порядке первого появления терма в корпусе).
static void sort_by_freq(uint32_t *a, size_t n) {
    uint32_t *tmp = (uint32_t *)malloc(n * sizeof(uint32_t));
    uint32_t *src = a, *dst = tmp;
    for (size_t width = 1; width < n; width *= 2) {
        for (size_t lo = 0; lo < n; lo += 2 * width) {
            size_t mid = lo + width < n ? lo + width : n, hi = lo + 2 * width < n ? lo + 2 * width : n;
            size_t i = lo, j = mid, k = lo;
            while (i < mid && j < hi) dst[k++] = g_freq[src[j]] > g_freq[src[i]] ? src[j++] : src[i++];
            while (i < mid) dst[k++] = src[i++];
            while (j < hi) dst[k++] = src[j++];
        }
        uint32_t *t = src; src = dst; dst = t;
    }
    if (src != a) memcpy(a, src, n * sizeof(uint32_t));
    free(tmp);
}

Хеш-таблица термов (используется везде: частоты, индекс): открытая адресация с линейным пробированием, хеш FNV-1a, заполнение не больше 50 % — затем удвоение (rehash):

lab_boolean/src/common.h — TermTable строки 165–209
// ---------------------------------------------------------------- хеш-таблица термов
// Открытая адресация с линейным пробированием; строки термов хранятся в общем пуле.
struct TermTable {
    struct Slot { uint32_t hash, id; };
    Slot *slots = nullptr;
    size_t cap = 0, size = 0;
    Vec<char> pool;            // байты всех термов подряд
    Vec<uint32_t> off;         // off[id]  — смещение терма в пуле
    Vec<uint8_t> len;          // len[id]  — длина терма в байтах

    static uint32_t fnv1a(const char *s, int n) {
        uint32_t h = 2166136261u;
        for (int i = 0; i < n; i++) { h ^= (unsigned char)s[i]; h *= 16777619u; }
        return h;
    }
    void rehash(size_t newcap) {
        Slot *ns = (Slot *)malloc(newcap * sizeof(Slot));
        for (size_t i = 0; i < newcap; i++) ns[i].id = UINT32_MAX;
        for (size_t i = 0; i < cap; i++) {
            if (slots[i].id == UINT32_MAX) continue;
            size_t j = slots[i].hash & (newcap - 1);
            while (ns[j].id != UINT32_MAX) j = (j + 1) & (newcap - 1);
            ns[j] = slots[i];
        }
        free(slots); slots = ns; cap = newcap;
    }
    // Возвращает id терма; если терма ещё нет — добавляет. isNew = true для нового.
    uint32_t get_or_add(const char *s, int n, bool &isNew) {
        if (size * 2 >= cap) rehash(cap ? cap * 2 : 1 << 16);
        uint32_t h = fnv1a(s, n);
        size_t j = h & (cap - 1);
        while (slots[j].id != UINT32_MAX) {
            uint32_t id = slots[j].id;
            if (slots[j].hash == h && len[id] == n && memcmp(pool.a + off[id], s, n) == 0) {
                isNew = false; return id;
            }
            j = (j + 1) & (cap - 1);
        }
        uint32_t id = (uint32_t)size++;
        slots[j].hash = h; slots[j].id = id;
        off.push((uint32_t)pool.n); len.push((uint8_t)n); pool.append(s, n);
        isNew = true;
        return id;
    }
};

4. Подбор констант Мандельброта#

Логарифмируем: $\log f = \log P - B \log(r + \rho)$ — при фиксированном ρ это линейная регрессия $\log f$ по $\log(r+\rho)$. Перебираем ρ на сетке и для каждого считаем МНК-прямую, выбираем ρ с наименьшей ошибкой. Две тонкости:

  • точки берутся равномерно в логарифмической шкале рангов (иначе миллионы точек хвоста «перевесят» начало);
  • хвост с частотами 1–2 исключается: там частоты — целые «ступеньки», и они искажают наклон.
lab_boolean/analysis/zipf.py — linreg строки 49–54
def linreg(xs, ys):
    mx, my = sum(xs) / len(xs), sum(ys) / len(ys)
    b = sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / sum((x - mx) ** 2 for x in xs)
    a = my - b * mx
    err = sum((y - a - b * x) ** 2 for x, y in zip(xs, ys)) / len(xs)
    return a, b, err
lab_boolean/analysis/zipf.py — mandelbrot строки 57–69
def mandelbrot(freqs):
    """Подбор P, ρ, B по точкам, равномерным в log r (иначе регрессию задавил бы хвост из единиц).
    Хвост с частотами 1–2 (горизонтальные «ступеньки» из-за дискретности частот) в подгонку не входит."""
    last = max(i for i, f in enumerate(freqs, 1) if f >= 3)
    ranks = log_sample(last)
    best = None
    for rho in [x / 2 for x in range(0, 401)]:             # ρ от 0 до 200 с шагом 0,5
        xs = [math.log(r + rho) for r in ranks]
        ys = [math.log(freqs[r - 1]) for r in ranks]
        a, b, err = linreg(xs, ys)
        if best is None or err < best[3]:
            best = (math.exp(a), rho, -b, err)
    return best

6. Причины расхождения с законом Ципфа#

  1. Двуязычный корпус. Служебные слова двух языков (the, и, в, and, of) образуют «полку» в начале: ранги 2–20 почти одинаково частые.
  2. Средние ранги выше прямой (факт в 2–2,5 раза больше C/r): смешение двух языков «раздвигает» ранги — каждый язык сам по себе ближе к Ципфу.
  3. Хвост — ступеньки: частоты целые, корпус конечен; множество слов с частотой 1, 2, 3.
  4. Нет лемматизации: формы одного слова считаются разными словами и дробят частоты (со стеммингом термов меньше: 386 882 основы вместо 642 124 словоформ).
  5. Тематичность: слова темы («железной», «railway», «станция») частее, чем в общем языке.

7. Зачем это поисковику#

  • Частые слова — кандидаты в стоп-слова и самые длинные списки документов в индексе (их выгодно сжимать — следующие работы).
  • Редкие слова — основная часть словаря (42,5 % — hapax), они определяют его размер.
  • По закону можно оценить размер словаря и индекса для большего корпуса. Родственный закон Хипса: словарь растёт как $V = k N^{\beta}$, β ≈ 0,4–0,6 — медленнее числа токенов.

10. Вопросы на защите#

  • Сформулируйте закон Ципфа. f·r ≈ const; в log-log — прямая с наклоном −1.
  • Чем Мандельброт лучше? Параметр ρ описывает начало списка, B — наклон.
  • Как подбирали константы? Логарифм → линейная регрессия по log(r+ρ), перебор ρ, точки равномерно в log-шкале, без хвоста f ≤ 2.
  • Почему расхождения? Два языка, ступеньки хвоста, нет лемматизации, тематичность.
  • Что такое hapax legomena? Слова, встретившиеся один раз: у меня 42,5 % словаря.
  • Как сортировали? Слиянием (стабильно, O(n log n) гарантированно).

// Токенизация — та же, что в индексаторе (next_token из common.h); с --stem считаются основы.
//
// Запуск: termfreq corpus.tsv freq.tsv [--stem]

static const uint64_t *g_freq;            // частоты — для сравнения при сортировке

// Сортировка слиянием номеров термов по убыванию частоты (стабильная: при равной частоте —
// в порядке первого появления терма в корпусе).
static void sort_by_freq(uint32_t *a, size_t n) {
    uint32_t *tmp = (uint32_t *)malloc(n * sizeof(uint32_t));
    uint32_t *src = a, *dst = tmp;
    for (size_t width = 1; width < n; width *= 2) {
        for (size_t lo = 0; lo < n; lo += 2 * width) {
            size_t mid = lo + width < n ? lo + width : n, hi = lo + 2 * width < n ? lo + 2 * width : n;
            size_t i = lo, j = mid, k = lo;
            while (i < mid && j < hi) dst[k++] = g_freq[src[j]] > g_freq[src[i]] ? src[j++] : src[i++];
            while (i < mid) dst[k++] = src[i++];
            while (j < hi) dst[k++] = src[j++];
        }
        uint32_t *t = src; src = dst; dst = t;
    }
    if (src != a) memcpy(a, src, n * sizeof(uint32_t));
    free(tmp);
}

def termfreq(corpus, out, stem=False):
    r = subprocess.run([TF, corpus, out] + (['--stem'] if stem else []), capture_output=True, encoding='utf-8')
    stats = {k: float(v) for k, v in (l.split() for l in r.stdout.strip().split('\n'))}
    rows = []
    with open(out, encoding='utf-8') as f:
        for line in f:
            rank, freq, term = line.rstrip('\n').split('\t')
            rows.append((int(rank), int(freq), term))
    return stats, rows


def log_sample(n, k=400):
    """~k рангов, равномерно распределённых по логарифмической шкале (для регрессии и графика)."""
    out, x = [], 1.0
    step = math.exp(math.log(n) / k)
    while x <= n:
        r = int(round(x))
        if not out or r != out[-1]:
            out.append(r)
        x *= step
    if out[-1] != n:
        out.append(n)
    return out


def linreg(xs, ys):
    mx, my = sum(xs) / len(xs), sum(ys) / len(ys)
    b = sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / sum((x - mx) ** 2 for x in xs)
    a = my - b * mx
    err = sum((y - a - b * x) ** 2 for x, y in zip(xs, ys)) / len(xs)
    return a, b, err


def mandelbrot(freqs):
    """Подбор P, ρ, B по точкам, равномерным в log r (иначе регрессию задавил бы хвост из единиц).
    Хвост с частотами 1–2 (горизонтальные «ступеньки» из-за дискретности частот) в подгонку не входит."""
    last = max(i for i, f in enumerate(freqs, 1) if f >= 3)
    ranks = log_sample(last)
    best = None
    for rho in [x / 2 for x in range(0, 401)]:             # ρ от 0 до 200 с шагом 0,5
        xs = [math.log(r + rho) for r in ranks]
        ys = [math.log(freqs[r - 1]) for r in ranks]
        a, b, err = linreg(xs, ys)
        if best is None or err < best[3]:
            best = (math.exp(a), rho, -b, err)
    return best


def split_corpus():

    d = tempfile.mkdtemp()

    with open(CORPUS, encoding='utf-8', newline='\n') as f:
        for line in f:
            files[line.split('\t', 1)[0]].write(line)
    for fh in files.values():
        fh.close()
    return {s: os.path.join(d, s + '.tsv') for s in NAMES}, d

//   lab5              — меню
//   lab5 top 20       — самые частые термы: ранг, частота, f·r, предсказание Ципфа C/r
//   lab5 word вагон   — ранг и частота слова, сравнение с законом
//   lab5 summary      — токенов, термов, hapax legomena, доля топ-100, константы

//   lab5 plot         — график частота–ранг в логарифмическом масштабе (в консоли)

#include "../src/common.h"

struct Row { unsigned long long f; std::string t; };
static std::vector<Row> g_rows;

static bool load() {
    if (!g_rows.empty()) return true;
    tool::each_line(tool::path("data/freq.tsv"), [](std::string &l) {
        size_t a = l.find('\t'), b = l.find('\t', a + 1);
        if (b != std::string::npos) g_rows.push_back({strtoull(l.c_str() + a + 1, nullptr, 10), l.substr(b + 1)});
        return true;
    });
    if (g_rows.empty()) printf("нет data/freq.tsv — выполните: lab5 recount\n");
    return !g_rows.empty();
}

// МНК для прямой y = a + b·x; возвращает среднеквадратичную ошибку.
static double linreg(const std::vector<double> &x, const std::vector<double> &y, double &a, double &b) {
    double n = (double)x.size(), sx = 0, sy = 0, sxx = 0, sxy = 0;
    for (size_t i = 0; i < x.size(); i++) { sx += x[i]; sy += y[i]; sxx += x[i] * x[i]; sxy += x[i] * y[i]; }
    b = (n * sxy - sx * sy) / (n * sxx - sx * sx);
    a = (sy - b * sx) / n;
    double e = 0;
    for (size_t i = 0; i < x.size(); i++) { double d = a + b * x[i] - y[i]; e += d * d; }
    return e / n;
}

// Закон Мандельброта f = P / (r + ρ)^B: перебор ρ, для каждого — МНК в логарифмах.
// Точки берутся равномерно по log r, хвост с частотами 1–2 не учитывается.
static void mandelbrot(double &P, double &rho, double &B) {
    size_t last = 0;
    for (size_t i = 0; i < g_rows.size(); i++) if (g_rows[i].f >= 3) last = i + 1;
    std::vector<size_t> ranks;
    for (int k = 0; k < 400; k++) {
        size_t r = (size_t)std::lround(std::exp(std::log((double)last) * k / 399.0));
        if (ranks.empty() || ranks.back() != r) ranks.push_back(r);
    }
    double best = 1e300;
    for (int i = 0; i <= 400; i++) {
        double rh = i / 2.0, a, b;
        std::vector<double> xs, ys;
        for (size_t r : ranks) { xs.push_back(std::log(r + rh)); ys.push_back(std::log((double)g_rows[r - 1].f)); }
        double e = linreg(xs, ys, a, b);
        if (e < best) { best = e; P = std::exp(a); rho = rh; B = -b; }
    }
}

static void cmd_top(const std::string &arg) {
    if (!load()) return;
    size_t n = arg.empty() ? 20 : (size_t)atoi(arg.c_str());
    double C = (double)g_rows[0].f;
    printf("  %s  %s %s %s %s\n", tool::lpad("ранг", 5).c_str(), tool::pad("терм", 20).c_str(), tool::lpad("частота", 12).c_str(), tool::lpad("f·r", 12).c_str(), tool::lpad("Ципф C/r", 12).c_str());
    for (size_t r = 1; r <= n && r <= g_rows.size(); r++)
        printf("  %5zu  %s %12s %12s %12s\n", r, tool::pad(g_rows[r - 1].t, 20).c_str(), tool::num(g_rows[r - 1].f).c_str(),
               tool::num(g_rows[r - 1].f * r).c_str(), tool::num((unsigned long long)(C / r)).c_str());
    printf("\nПо закону Ципфа f·r ≈ const (= C, частота первого терма).\n");
}

static void cmd_word(const std::string &arg) {
    if (!load()) return;
    std::string w = arg.empty() ? "вагон" : arg;
    char tok[MAX_TERM_BYTES + 4]; size_t pos = 0; int chars;
    int len = next_token(w.data(), w.size(), pos, tok, chars);
    std::string t(tok, len > 0 ? len : 0);
    for (size_t i = 0; i < g_rows.size(); i++)
        if (g_rows[i].t == t) {
            double C = (double)g_rows[0].f; size_t r = i + 1;
            printf("  терм «%s»: ранг %s, частота %s\n  Ципф предсказывает C/r = %.0f (отношение %.2f)\n",
                   t.c_str(), tool::num(r).c_str(), tool::num(g_rows[i].f).c_str(), C / r, g_rows[i].f / (C / r));
            return;
        }
    printf("  терм «%s» не встречается в корпусе\n", t.c_str());
}

static void cmd_summary(const std::string &) {
    if (!load()) return;
    unsigned long long tokens = 0, top100 = 0; size_t hapax = 0;
    for (size_t i = 0; i < g_rows.size(); i++) {
        tokens += g_rows[i].f;
        if (i < 100) top100 += g_rows[i].f;
        if (g_rows[i].f == 1) hapax++;
    }
    double P, rho, B; mandelbrot(P, rho, B);
    printf("  токенов (вхождений)        %s\n", tool::num(tokens).c_str());
    printf("  различных термов           %s\n", tool::num(g_rows.size()).c_str());
    printf("  hapax legomena (частота 1) %s (%.1f%% словаря)\n", tool::num(hapax).c_str(), 100.0 * hapax / g_rows.size());
    printf("  доля 100 самых частых      %.1f%% всех вхождений\n", 100.0 * top100 / tokens);
    printf("  константа Ципфа C = f(1)   %s\n", tool::num(g_rows[0].f).c_str());
    printf("  Мандельброт: P = %.0f, ρ = %.1f, B = %.3f\n", P, rho, B);
}

static void cmd_ranks(const std::string &) {

    double C = (double)g_rows[0].f, P, rho, B; mandelbrot(P, rho, B);
    printf("  %s  %s %s %s %s\n", tool::lpad("ранг", 9).c_str(), tool::pad("терм", 18).c_str(), tool::lpad("частота", 12).c_str(), tool::lpad("Ципф", 12).c_str(), tool::lpad("Мандельброт", 12).c_str());
    for (size_t r = 1; r <= g_rows.size(); r *= 10)
        printf("  %s  %s %s %12.0f %12.0f\n", tool::lpad(tool::num(r), 9).c_str(), tool::pad(g_rows[r - 1].t, 18).c_str(),

    printf("\nНа малых рангах Ципф завышает частоту; Мандельброт с поправкой ρ ближе к данным.\n");
}

// График log f от log r символами: * — данные, + — Ципф C/r.
static void cmd_plot(const std::string &) {
    if (!load()) return;
    const int W = 70, H = 22;
    double lr = std::log10((double)g_rows.size()), lf = std::log10((double)g_rows[0].f);
    std::vector<std::string> g(H, std::string(W, ' '));
    for (int x = 0; x < W; x++) {
        double r = std::pow(10.0, lr * x / (W - 1));
        size_t ri = (size_t)r; if (ri < 1) ri = 1; if (ri > g_rows.size()) ri = g_rows.size();
        int yd = (int)std::lround((H - 1) * (1 - std::log10((double)g_rows[ri - 1].f) / lf));
        double zipf = g_rows[0].f / r;
        int yz = zipf >= 1 ? (int)std::lround((H - 1) * (1 - std::log10(zipf) / lf)) : -1;
        if (yz >= 0 && yz < H) g[yz][x] = '+';
        if (yd >= 0 && yd < H) g[yd][x] = '*';
    }
    printf("  log f\n");
    for (auto &l : g) printf("  |%s\n", l.c_str());
    printf("  +%s log r\n  * — частоты корпуса, + — закон Ципфа C/r (оси логарифмические)\n", std::string(W, '-').c_str());
}
