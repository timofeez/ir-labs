ЛР4. Стемминг#

Задание (lesson-04/homework.md, вариант на «3» — стемминг): добавить в поисковую систему поиск без учёта словоформ. Оценить качество поиска после внедрения: стало ли лучше? Привести запросы, где стало хуже, объяснить причину и предложить, как исправить, не ухудшив остальные.

1. Проблема словоформ#

У русского слова — десятки форм: электровоз, электровоза, электровозу, электровозом, электровозе, электровозы, электровозов, электровозам, электровозами, электровозах. Пользователь пишет одну форму, а в документах — другие. Без обработки поиск пропускает документы — падает полнота.

Пример из моего корпуса: запрос «машинисту поезда» без стемминга находит 204 документа, а релевантных (содержащих оба слова в любой форме) — 2 353. Полнота 0,087.

Я сделал стемминг (вариант на «3»): русский — Snowball (развитие алгоритма Портера), английский — классический алгоритм Портера (1980). Язык определяется по буквам терма: только кириллица → русский, только a–z → английский, иначе (цифры, смешанные) — терм не меняется.

3. Алгоритм Snowball для русского#

  1. Области слова. RV — часть после первой гласной; R1 — после первой согласной, идущей за гласной; R2 — то же внутри R1. Окончание отрезается, только если целиком лежит в нужной области — это не даёт «съесть» корень коротких слов.
  2. Шаг 1. Отрезать деепричастие (вши, ивши…); если нет — возвратное ся/сь, затем прилагательное (ого, ыми…) с причастием перед ним, иначе глагол (ила, ешь…), иначе существительное (ами, ов, ия…). Всегда берётся самое длинное подходящее окончание.
  3. Шаг 2. Отрезать и на конце.
  4. Шаг 3. Словообразовательный суффикс ост/ость — только в R2.
  5. Шаг 4. Превосходная степень ейш/ейше, двойное нн → н, мягкий знак.

Некоторые окончания (группа 1) допускаются только после а или я (чита-вший, но не пи-вший).

// Самое длинное окончание из таблицы на конце w[0..n), начинающееся не раньше lim. Возвращает его номер или -1.
static int longest(const Table &t, const uint32_t *w, int n, int lim) {
    int best = -1;
    for (int k = 0; k < t.n; k++) {
        const Suf &s = t.s[k];
        if (s.n > n - lim) continue;
        if (memcmp(w + n - s.n, s.c, s.n * sizeof(uint32_t)) != 0) continue;
        if (best < 0 || s.n > t.s[best].n) best = k;
    }
    return best;
}

static int stem_ru(uint32_t *w, int n) {
    ru_init();
    // Области: RV — после первой гласной; R2 — область R1 внутри R1 (R1 — после первой согласной, идущей за гласной)
    int pV = n, p2 = n, i = 0;
    while (i < n && !ru_vowel(w[i])) i++;
    if (i < n) {
        pV = i + 1; i = pV;
        while (i < n && ru_vowel(w[i])) i++;          // до первой согласной
        if (i < n) {
            i++;                                       // i = p1
            while (i < n && !ru_vowel(w[i])) i++;
            if (i < n) {
                i++;
                while (i < n && ru_vowel(w[i])) i++;
                if (i < n) p2 = i + 1;
            }
        }
    }
    // Шаг 1: деепричастие, либо (возвратное «ся» + прилагательное/причастие, глагол или существительное)
    if (!cut(PERF, w, n, pV)) {
        cut(REFL, w, n, pV);
        int k = longest(ADJ, w, n, pV);
        if (k >= 0) {
            n -= ADJ.s[k].n;
            cut(PART, w, n, pV);                       // причастие перед окончанием прилагательного
        } else if (!cut(VERB, w, n, pV)) {
            cut(NOUN, w, n, pV);
        }
    }
    // Шаг 2: «и» на конце
    if (n - 1 >= pV && w[n - 1] == RU_I) n--;
    // Шаг 3: словообразовательный суффикс «ост(ь)» в R2
    int k = longest(DERIV, w, n, p2 > pV ? p2 : pV);
    if (k >= 0) n -= DERIV.s[k].n;
    // Шаг 4: превосходная степень «ейш(е)», двойное «нн» -> «н», мягкий знак
    k = longest(SUPER, w, n, pV);
    if (k >= 0) {
        n -= SUPER.s[k].n;
        if (n - 2 >= pV && w[n - 1] == RU_N && w[n - 2] == RU_N) n--;
    } else if (n - 2 >= pV && w[n - 1] == RU_N && w[n - 2] == RU_N) {
        n--;
    } else if (n - 1 >= pV && w[n - 1] == RU_SOFT) {
        n--;
    }
    return n;
}

4. Алгоритм Портера для английского#

Слово рассматривается как [C](VC)^m[V] (C — серия согласных, V — гласных), m — «мера». Шаги: 1a — множественное число (sses → ss, ies → i, s →); 1b — -ed, -ing (с восстановлением e: hoping → hope); 1c — y → i; 2–4 — суффиксы (ational → ate, ness, ment…) при условии на меру; 5 — конечное e и ll.

static int stem_en(char *w, int n) {
    if (n <= 0) return n;
    // «y» — согласная в начале слова и после гласной (помечается как 'Y')
    if (w[0] == 'y') w[0] = 'Y';
    for (int i = 1; i < n; i++) if (w[i] == 'y' && en_v(w[i - 1])) w[i] = 'Y';
    // Области R1, R2
    int p1 = n, p2 = n, i = 0;
    while (i < n && !en_v(w[i])) i++;
    if (i < n) {
        i++; while (i < n && en_v(w[i])) i++;
        if (i < n) {
            p1 = i + 1; i = p1;
            while (i < n && !en_v(w[i])) i++;
            if (i < n) { i++; while (i < n && en_v(w[i])) i++; if (i < n) p2 = i + 1; }
        }
    }
    // Шаг 1a
    if (ends(w, n, "sses")) n -= 2;
    else if (ends(w, n, "ies")) n -= 2;
    else if (ends(w, n, "ss")) {}
    else if (ends(w, n, "s")) n -= 1;
    // Шаг 1b
    if (ends(w, n, "eed")) {
        if (n - 3 >= p1) n -= 1;
    } else {
        int L = ends(w, n, "ing") ? 3 : ends(w, n, "ed") ? 2 : 0;
        if (L && has_vowel(w, n - L)) {
            n -= L;
            if (ends(w, n, "at") || ends(w, n, "bl") || ends(w, n, "iz")) w[n++] = 'e';
            else if (n >= 2 && w[n - 1] == w[n - 2] && strchr("bdfgmnprt", w[n - 1])) n--;
            else if (n == p1 && shortv(w, n)) w[n++] = 'e';
        }
    }
    // Шаг 1c
    if (n >= 1 && (w[n - 1] == 'y' || w[n - 1] == 'Y') && has_vowel(w, n - 1)) w[n - 1] = 'i';
    // Шаг 2 (в R1)
    static const Rule S2[] = {
        {"tional", "tion"}, {"enci", "ence"}, {"anci", "ance"}, {"abli", "able"}, {"entli", "ent"},
        {"eli", "e"}, {"izer", "ize"}, {"ization", "ize"}, {"ational", "ate"}, {"ation", "ate"},
        {"ator", "ate"}, {"alli", "al"}, {"alism", "al"}, {"aliti", "al"}, {"fulness", "ful"},
        {"ousli", "ous"}, {"ousness", "ous"}, {"iveness", "ive"}, {"iviti", "ive"}, {"biliti", "ble"}};
    int k = en_longest(S2, sizeof(S2) / sizeof(S2[0]), w, n);
    if (k >= 0) { int L = (int)strlen(S2[k].suf); if (n - L >= p1) replace_end(w, n, L, S2[k].rep); }
    // Шаг 3 (в R1)
    static const Rule S3[] = {
        {"alize", "al"}, {"icate", "ic"}, {"iciti", "ic"}, {"ical", "ic"}, {"ative", ""}, {"ful", ""}, {"ness", ""}};
    k = en_longest(S3, sizeof(S3) / sizeof(S3[0]), w, n);
    if (k >= 0) { int L = (int)strlen(S3[k].suf); if (n - L >= p1) replace_end(w, n, L, S3[k].rep); }
    // Шаг 4 (в R2)
    static const Rule S4[] = {
        {"al", ""}, {"ance", ""}, {"ence", ""}, {"er", ""}, {"ic", ""}, {"able", ""}, {"ible", ""},
        {"ant", ""}, {"ement", ""}, {"ment", ""}, {"ent", ""}, {"ou", ""}, {"ism", ""}, {"ate", ""},
        {"iti", ""}, {"ous", ""}, {"ive", ""}, {"ize", ""}, {"ion", ""}};
    k = en_longest(S4, sizeof(S4) / sizeof(S4[0]), w, n);
    if (k >= 0) {
        int L = (int)strlen(S4[k].suf);
        if (n - L >= p2) {
            if (strcmp(S4[k].suf, "ion") != 0) n -= L;
            else if (n - L - 1 >= 0 && (w[n - L - 1] == 's' || w[n - L - 1] == 't')) n -= L;
        }
    }
    // Шаг 5a: конечное «e»
    if (n >= 1 && w[n - 1] == 'e') {
        int c = n - 1;
        if (c >= p2 || (c >= p1 && !shortv(w, c))) n--;
    }
    // Шаг 5b: «ll» -> «l» в R2
    if (n >= 2 && w[n - 1] == 'l' && n - 1 >= p2 && w[n - 2] == 'l') n--;
    for (int j = 0; j < n; j++) if (w[j] == 'Y') w[j] = 'y';
    return n;
}

// ---------------------------------------------------------------- русский (Snowball)
struct Suf { uint32_t c[8]; int n; int grp; };       // grp = 1: перед окончанием должна стоять «а» или «я»
struct Table { Suf s[48]; int n; };

static void add(Table &t, const char *u, int grp) {
    Suf &x = t.s[t.n++];
    size_t i = 0, n = strlen(u);
    x.n = 0; x.grp = grp;
    while (i < n) x.c[x.n++] = utf8_decode((const unsigned char *)u, n, i);
}
static void add_all(Table &t, const char *list, int grp) {     // слова через пробел
    char buf[32]; int k = 0;
    for (const char *p = list;; p++) {
        if (*p == ' ' || *p == 0) { if (k) { buf[k] = 0; add(t, buf, grp); k = 0; } if (!*p) break; }
        else buf[k++] = *p;
    }
}

const uint32_t RU_A = 0x430, RU_YA = 0x44F, RU_I = 0x438, RU_N = 0x43D, RU_SOFT = 0x44C;

static bool ru_vowel(uint32_t c) {         // а е и о у ы э ю я
    return c == 0x430 || c == 0x435 || c == 0x438 || c == 0x43E || c == 0x443 ||

// Самое длинное окончание из таблицы на конце w[0..n), начинающееся не раньше lim. Возвращает его номер или -1.
static int longest(const Table &t, const uint32_t *w, int n, int lim) {
    int best = -1;
    for (int k = 0; k < t.n; k++) {
        const Suf &s = t.s[k];
        if (s.n > n - lim) continue;
        if (memcmp(w + n - s.n, s.c, s.n * sizeof(uint32_t)) != 0) continue;
        if (best < 0 || s.n > t.s[best].n) best = k;
    }
    return best;

// Как «among ... delete» в Snowball: берётся самое длинное окончание; если для него не выполнено
// условие группы 1 (перед ним «а»/«я»), шаг не выполняется (более короткие не пробуются).
static bool cut(const Table &t, const uint32_t *w, int &n, int lim) {
    int k = longest(t, w, n, lim);
    if (k < 0) return false;
    int L = t.s[k].n;
    if (t.s[k].grp == 1) {
        int p = n - L - 1;
        if (p < lim || (w[p] != RU_A && w[p] != RU_YA)) return false;
    }
    n -= L;
    return true;
}

// ---------------------------------------------------------------- английский (Портер)
static bool en_v(char c) { return c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' || c == 'y'; }

static bool ends(const char *w, int n, const char *s) {
    int L = (int)strlen(s);
    return L <= n && memcmp(w + n - L, s, L) == 0;
}
// «Короткий слог» на конце w[0..n): согласная (не w, x, Y), перед ней гласная, перед ней согласная
static bool shortv(const char *w, int n) {
    if (n < 3) return false;
    char c = w[n - 1];
    return !en_v(c) && c != 'w' && c != 'x' && c != 'Y' && en_v(w[n - 2]) && !en_v(w[n - 3]);
}
static bool has_vowel(const char *w, int n) {
    for (int i = 0; i < n; i++) if (en_v(w[i])) return true;
    return false;
}

struct Rule { const char *suf; const char *rep; };

// Самое длинное правило из списка, подходящее к концу слова; -1, если ни одно не подходит
static int en_longest(const Rule *r, int cnt, const char *w, int n) {
    int best = -1, bl = -1;
    for (int k = 0; k < cnt; k++) {
        int L = (int)strlen(r[k].suf);
        if (L > bl && ends(w, n, r[k].suf)) { best = k; bl = L; }
    }
    return best;
}
static void replace_end(char *w, int &n, int cut_len, const char *rep) {
    n -= cut_len;
    int L = (int)strlen(rep);
    memcpy(w + n, rep, L);
    n += L;
}

    else if (ends(w, n, "s")) n -= 1;
    // Шаг 1b
    if (ends(w, n, "eed")) {
        if (n - 3 >= p1) n -= 1;
    } else {
        int L = ends(w, n, "ing") ? 3 : ends(w, n, "ed") ? 2 : 0;

            else if (n >= 2 && w[n - 1] == w[n - 2] && strchr("bdfgmnprt", w[n - 1])) n--;
            else if (n == p1 && shortv(w, n)) w[n++] = 'e';
        }

        {"ant", ""}, {"ement", ""}, {"ment", ""}, {"ent", ""}, {"ou", ""}, {"ism", ""}, {"ate", ""},
        {"iti", ""}, {"ous", ""}, {"ive", ""}, {"ize", ""}, {"ion", ""}};
    k = en_longest(S4, sizeof(S4) / sizeof(S4[0]), w, n);
    if (k >= 0) {
        int L = (int)strlen(S4[k].suf);
        if (n - L >= p2) {
            if (strcmp(S4[k].suf, "ion") != 0) n -= L;
            else if (n - L - 1 >= 0 && (w[n - L - 1] == 's' || w[n - L - 1] == 't')) n -= L;
        }
    }
    // Шаг 5a: конечное «e»
    if (n >= 1 && w[n - 1] == 'e') {
        int c = n - 1;
        if (c >= p2 || (c >= p1 && !shortv(w, c))) n--;
    }
    // Шаг 5b: «ll» -> «l» в R2
    if (n >= 2 && w[n - 1] == 'l' && n - 1 >= p2 && w[n - 2] == 'l') n--;
    for (int j = 0; j < n; j++) if (w[j] == 'Y') w[j] = 'y';
    return n;
}

// Основа терма s[0..n) (UTF-8, нижний регистр) -> out; возвращает длину основы в байтах.
static int stem(const char *s, int n, char *out) {
    bool en = n > 0, ru = n > 0;
    uint32_t cp[MAXW]; int m = 0;
    size_t i = 0;
    while (i < (size_t)n && m < MAXW) {
        uint32_t c = utf8_decode((const unsigned char *)s, n, i);
        cp[m++] = c;
        if (!(c >= 'a' && c <= 'z')) en = false;
        if (!(c >= 0x430 && c <= 0x44F)) ru = false;
    }
    if (en) {
        memcpy(out, s, n);
        return stem_en(out, n);
    }
    if (ru) {
        m = stem_ru(cp, m);
        int len = 0;
        for (int j = 0; j < m; j++) len += utf8_encode(cp[j], out + len);
        return len;
    }
    memcpy(out, s, n);
    return n;
}

6. Оценка качества#

Метрики (поиск булев, без ранжирования, поэтому DCG/NDCG неприменимы):

  • Точность P = |найдено ∩ релевантно| / |найдено|;
  • Полнота R = |найдено ∩ релевантно| / |релевантно|;
  • F1 = 2PR / (P + R) — среднее гармоническое: маленькое, если мала хотя бы одна из метрик.

Эталон (что считать релевантным): документ содержит все слова запроса в любой форме слова. Для каждого слова вручную составлен список его форм (лемма → формы), документы найдены перебором по всему корпусу. Запросов — 23: русские и английские, частые и редкие.

P R F1
Без стемминга 1,000 0,551 0,668
Со стеммингом 0,921 0,968 0,914
С исправлениями (Porter2 + исключения) 0,988 0,968 0,968

Стало лучше в 18 запросах из 23, хуже — в 4 (news, general, university, «депо»), без изменений — 1. Рост — за счёт полноты; точность немного упала — это типичный компромисс.

7. Где стало хуже и почему#

  • Пересечение (overstemming) — разные слова получают одну основу → падает точность: news → new (находит «new»), general и generation → gener, university и universe → univers, «метро» и «метров» → метр.
  • Недостаточный стемминг (understemming) — формы одного слова дают разные основы → падает полнота: «автомобили» → автомоб, «автомобиль» → автомобил.

Как исправить, не ухудшив остальное: Porter2 (Snowball English) со списком исключений (news, generally…); не стеммировать несклоняемые слова («депо», «метро», «кино»); словарная лемматизация для русского; в ранжированном поиске — бонус за точное совпадение формы. Проверено расчётом: F1 0,914 → 0,968, ни один запрос не стал хуже.

10. Вопросы на защите#

  • Стемминг или лемматизация — что сделано и почему? Стемминг: на «3» требуется он; без словаря, быстро.
  • Как работает Snowball? Области RV/R1/R2, самое длинное окончание из таблиц по шагам.
  • Почему основа не слово? Стеммер не знает словаря — ему важно, чтобы формы совпали, а не чтобы основа была словом.
  • Как проверял правильность? Сверка со snowballstemmer на всех 564 тыс. термах — 0 расхождений.
  • Как встроено в поиск? Индекс основ + флаг в заголовке; слова запроса стеммируются тем же стеммером.
  • Почему точность упала? Пересечения: news → new, «метро» и «метров» → метр.

// Стемминг (ЛР «Стемминг»): отбрасывание окончаний и суффиксов, чтобы разные словоформы
// одного слова давали одну основу («электровозы», «электровозов» -> «электровоз»).
//   русский    — алгоритм Портера в версии Snowball (snowballstem.org/algorithms/russian);
//   английский — классический алгоритм Портера (snowballstem.org/algorithms/porter).
// Язык определяется по буквам терма: только кириллица -> русский, только a-z -> английский,
// иначе (цифры, смешанные, другие алфавиты) терм не меняется. Без STL.
#pragma once
#include "common.h"

namespace stem {

const int MAXW = 64;                      // как MAX_TERM_BYTES: длиннее термов не бывает

// ---------------------------------------------------------------- русский (Snowball)
struct Suf { uint32_t c[8]; int n; int grp; };       // grp = 1: перед окончанием должна стоять «а» или «я»
struct Table { Suf s[48]; int n; };

static void add(Table &t, const char *u, int grp) {
    Suf &x = t.s[t.n++];
    size_t i = 0, n = strlen(u);
    x.n = 0; x.grp = grp;
    while (i < n) x.c[x.n++] = utf8_decode((const unsigned char *)u, n, i);
}
static void add_all(Table &t, const char *list, int grp) {     // слова через пробел
    char buf[32]; int k = 0;
    for (const char *p = list;; p++) {
        if (*p == ' ' || *p == 0) { if (k) { buf[k] = 0; add(t, buf, grp); k = 0; } if (!*p) break; }
        else buf[k++] = *p;
    }
}

const uint32_t RU_A = 0x430, RU_YA = 0x44F, RU_I = 0x438, RU_N = 0x43D, RU_SOFT = 0x44C;

static bool ru_vowel(uint32_t c) {         // а е и о у ы э ю я
    return c == 0x430 || c == 0x435 || c == 0x438 || c == 0x43E || c == 0x443 ||
           c == 0x44B || c == 0x44D || c == 0x44E || c == 0x44F;
}

// Самое длинное окончание из таблицы на конце w[0..n), начинающееся не раньше lim. Возвращает его номер или -1.
static int longest(const Table &t, const uint32_t *w, int n, int lim) {
    int best = -1;
    for (int k = 0; k < t.n; k++) {
        const Suf &s = t.s[k];
        if (s.n > n - lim) continue;
        if (memcmp(w + n - s.n, s.c, s.n * sizeof(uint32_t)) != 0) continue;
        if (best < 0 || s.n > t.s[best].n) best = k;
    }
    return best;
}
// Как «among ... delete» в Snowball: берётся самое длинное окончание; если для него не выполнено
// условие группы 1 (перед ним «а»/«я»), шаг не выполняется (более короткие не пробуются).
static bool cut(const Table &t, const uint32_t *w, int &n, int lim) {
    int k = longest(t, w, n, lim);
    if (k < 0) return false;
    int L = t.s[k].n;
    if (t.s[k].grp == 1) {
        int p = n - L - 1;
        if (p < lim || (w[p] != RU_A && w[p] != RU_YA)) return false;
    }
    n -= L;
    return true;
}

static int stem_ru(uint32_t *w, int n) {
    ru_init();
    // Области: RV — после первой гласной; R2 — область R1 внутри R1 (R1 — после первой согласной, идущей за гласной)
    int pV = n, p2 = n, i = 0;
    while (i < n && !ru_vowel(w[i])) i++;
    if (i < n) {
        pV = i + 1; i = pV;
        while (i < n && ru_vowel(w[i])) i++;          // до первой согласной
        if (i < n) {
            i++;                                       // i = p1
            while (i < n && !ru_vowel(w[i])) i++;
            if (i < n) {
                i++;
                while (i < n && ru_vowel(w[i])) i++;
                if (i < n) p2 = i + 1;
            }
        }
    }
    // Шаг 1: деепричастие, либо (возвратное «ся» + прилагательное/причастие, глагол или существительное)
    if (!cut(PERF, w, n, pV)) {
        cut(REFL, w, n, pV);
        int k = longest(ADJ, w, n, pV);
        if (k >= 0) {
            n -= ADJ.s[k].n;
            cut(PART, w, n, pV);                       // причастие перед окончанием прилагательного
        } else if (!cut(VERB, w, n, pV)) {
            cut(NOUN, w, n, pV);
        }
    }
    // Шаг 2: «и» на конце
    if (n - 1 >= pV && w[n - 1] == RU_I) n--;
    // Шаг 3: словообразовательный суффикс «ост(ь)» в R2
    int k = longest(DERIV, w, n, p2 > pV ? p2 : pV);
    if (k >= 0) n -= DERIV.s[k].n;
    // Шаг 4: превосходная степень «ейш(е)», двойное «нн» -> «н», мягкий знак
    k = longest(SUPER, w, n, pV);
    if (k >= 0) {
        n -= SUPER.s[k].n;
        if (n - 2 >= pV && w[n - 1] == RU_N && w[n - 2] == RU_N) n--;
    } else if (n - 2 >= pV && w[n - 1] == RU_N && w[n - 2] == RU_N) {
        n--;
    } else if (n - 1 >= pV && w[n - 1] == RU_SOFT) {
        n--;
    }
    return n;
}

static bool en_v(char c) { return c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' || c == 'y'; }

static bool ends(const char *w, int n, const char *s) {
    int L = (int)strlen(s);
    return L <= n && memcmp(w + n - L, s, L) == 0;
}
// «Короткий слог» на конце w[0..n): согласная (не w, x, Y), перед ней гласная, перед ней согласная
static bool shortv(const char *w, int n) {
    if (n < 3) return false;
    char c = w[n - 1];
    return !en_v(c) && c != 'w' && c != 'x' && c != 'Y' && en_v(w[n - 2]) && !en_v(w[n - 3]);
}
static bool has_vowel(const char *w, int n) {
    for (int i = 0; i < n; i++) if (en_v(w[i])) return true;
    return false;
}

struct Rule { const char *suf; const char *rep; };

// Самое длинное правило из списка, подходящее к концу слова; -1, если ни одно не подходит
static int en_longest(const Rule *r, int cnt, const char *w, int n) {
    int best = -1, bl = -1;
    for (int k = 0; k < cnt; k++) {
        int L = (int)strlen(r[k].suf);
        if (L > bl && ends(w, n, r[k].suf)) { best = k; bl = L; }
    }
    return best;
}
static void replace_end(char *w, int &n, int cut_len, const char *rep) {
    n -= cut_len;
    int L = (int)strlen(rep);
    memcpy(w + n, rep, L);
    n += L;
}

static int stem_en(char *w, int n) {
    if (n <= 0) return n;
    // «y» — согласная в начале слова и после гласной (помечается как 'Y')
    if (w[0] == 'y') w[0] = 'Y';
    for (int i = 1; i < n; i++) if (w[i] == 'y' && en_v(w[i - 1])) w[i] = 'Y';
    // Области R1, R2
    int p1 = n, p2 = n, i = 0;
    while (i < n && !en_v(w[i])) i++;
    if (i < n) {
        i++; while (i < n && en_v(w[i])) i++;
        if (i < n) {
            p1 = i + 1; i = p1;
            while (i < n && !en_v(w[i])) i++;
            if (i < n) { i++; while (i < n && en_v(w[i])) i++; if (i < n) p2 = i + 1; }
        }
    }
    // Шаг 1a
    if (ends(w, n, "sses")) n -= 2;
    else if (ends(w, n, "ies")) n -= 2;

        }

// Основа терма s[0..n) (UTF-8, нижний регистр) -> out; возвращает длину основы в байтах.
static int stem(const char *s, int n, char *out) {
    bool en = n > 0, ru = n > 0;
    uint32_t cp[MAXW]; int m = 0;
    size_t i = 0;
    while (i < (size_t)n && m < MAXW) {
        uint32_t c = utf8_decode((const unsigned char *)s, n, i);
        cp[m++] = c;
        if (!(c >= 'a' && c <= 'z')) en = false;
        if (!(c >= 0x430 && c <= 0x44F)) ru = false;
    }
    if (en) {
        memcpy(out, s, n);
        return stem_en(out, n);
    }
    if (ru) {
        m = stem_ru(cp, m);
        int len = 0;
        for (int j = 0; j < m; j++) len += utf8_encode(cp[j], out + len);
        return len;
    }
    memcpy(out, s, n);
    return n;
}

def cmd_forms(arg):
    w = (arg or input('слово: ')).strip().lower()
    target = stems([w])[0]
    forms = []
    with open(data('freq.tsv'), encoding='utf-8') as f:
        cand = [(int(l.split('\t')[1]), l.split('\t')[2].strip()) for l in f
                if l.split('\t')[2].startswith(target[:max(3, len(target) - 1)])]
    cand_words = [c[1] for c in cand]
    st = stems(cand_words) if cand_words else []
    for (freq, word), s in zip(cand, st):
        if s == target:
            forms.append((word, freq))
    forms.sort(key=lambda x: -x[1])
    print(f'Основа «{target}»: {len(forms)} словоформ в корпусе, {fmt(sum(f for _, f in forms))} вхождений')
    table([(w, fmt(f)) for w, f in forms[:30]], ('словоформа', 'вхождений'))


def cmd_eval(_):

    d = json.load(open(p, encoding='utf-8'))

           for r in rows], ('запрос', 'эталон', 'без стем.', 'со стем.', 'F1 до', 'F1 после', 'F1 испр.'))

    print('\nF1 до — без стемминга, после — стемминг Snowball/Портер, испр. — с исправлениями из отчёта.')
