ЛР8. Булев поиск#

Задание (Задания-2021.pdf): поиск по булеву индексу. Синтаксис: пробел или && — И, || — ИЛИ, ! — НЕ, скобки. Парсер должен быть толерантным к лишним пробелам и операторам. Интерфейс в двух видах: веб-сервис (стартовая страница с формой и страница выдачи по 50 результатов со ссылкой на следующие 50) и утилита командной строки (запросы со стандартного ввода, результат — в стандартный вывод). В отчёте: скорость запросов, тяжёлые запросы, как проверялась корректность.

1. Булева модель поиска#

Запрос — логическое выражение из термов: Brutus AND Caesar AND NOT Calpurnia. Документ либо удовлетворяет выражению, либо нет — без степеней «насколько», без ранжирования.

  • Плюсы: проста, предсказуема (точно понятно, почему документ найден), быстра.
  • Минусы: нет ранжирования (все найденные «одинаково хороши»); либо слишком много результатов (ИЛИ), либо ноль (длинное И); пользователю трудно писать логические выражения.

2. Операции над отсортированными списками#

И (пересечение) — два указателя: равны → в ответ и сдвигаем оба; иначе сдвигаем меньший. O(m + n).

lab_boolean/src/search.cpp — op_and строки 80–88
// ---------------------------------------------------------------- операции над списками
static void op_and(const Vec<uint32_t> &a, const Vec<uint32_t> &b, Vec<uint32_t> &r) {
    r.clear(); size_t i = 0, j = 0;
    while (i < a.n && j < b.n) {
        if (a[i] < b[j]) i++;
        else if (a[i] > b[j]) j++;
        else { r.push(a[i]); i++; j++; }
    }
}

ИЛИ (объединение) — слияние двух отсортированных списков, O(m + n):

lab_boolean/src/search.cpp — op_or строки 89–98
static void op_or(const Vec<uint32_t> &a, const Vec<uint32_t> &b, Vec<uint32_t> &r) {
    r.clear(); r.reserve(a.n + b.n); size_t i = 0, j = 0;
    while (i < a.n && j < b.n) {
        if (a[i] < b[j]) r.push(a[i++]);
        else if (a[i] > b[j]) r.push(b[j++]);
        else { r.push(a[i]); i++; j++; }
    }
    while (i < a.n) r.push(a[i++]);
    while (j < b.n) r.push(b[j++]);
}

НЕ — дополнение до всех документов, O(N). Но A && !B выгоднее считать вычитанием (O(|A| + |B|)), не строя дополнение почти всего корпуса:

lab_boolean/src/search.cpp — op_and_not строки 99–106
static void op_and_not(const Vec<uint32_t> &a, const Vec<uint32_t> &b, Vec<uint32_t> &r) {
    r.clear(); size_t j = 0;
    for (size_t i = 0; i < a.n; i++) {
        while (j < b.n && b[j] < a[i]) j++;
        if (j < b.n && b[j] == a[i]) continue;
        r.push(a[i]);
    }
}
lab_boolean/src/search.cpp — op_not строки 107–113
static void op_not(const Vec<uint32_t> &a, uint32_t nDocs, Vec<uint32_t> &r) {
    r.clear(); size_t j = 0;
    for (uint32_t d = 0; d < nDocs; d++) {
        if (j < a.n && a[j] == d) { j++; continue; }
        r.push(d);
    }
}

Оптимизации: И — от коротких списков к длинным (промежуточные результаты маленькие); A && !B — вычитанием; для очень больших корпусов — skip pointers (прыжки по √L элементов списка при пересечении) — следующие работы.

3. Разбор запроса#

Лексер#

Строка → лексемы: (, ), !, &&/&, ||/| и слова. Слово прогоняется через тот же токенизатор, что документы (ЛР3); если индекс основ — ещё и через стеммер (ЛР4). Слово «ВЛ-80» даёт два терма, между ними вставляется неявное И:

lab_boolean/src/search.cpp — lex строки 119–148
static void lex(const Index &idx, const char *q, size_t n, Vec<QTok> &out) {
    out.clear();
    size_t i = 0;
    char tok[MAX_TERM_BYTES + 4], sb[MAX_TERM_BYTES + 8];
    while (i < n) {
        char c = q[i];
        if (c == ' ' || c == '\t' || c == '\r' || c == '\n') { i++; continue; }
        if (c == '(') { out.push({T_LP, 0}); i++; continue; }
        if (c == ')') { out.push({T_RP, 0}); i++; continue; }
        if (c == '!') { out.push({T_NOT, 0}); i++; continue; }
        if (c == '&') { while (i < n && q[i] == '&') i++; out.push({T_AND, 0}); continue; }
        if (c == '|') { while (i < n && q[i] == '|') i++; out.push({T_OR, 0}); continue; }
        // слово: всё до пробела или спецсимвола; внутри него работает тот же токенизатор, что и в индексаторе
        size_t st = i;
        while (i < n && !strchr(" \t\r\n()!&|", q[i])) i++;
        size_t pos = 0; int chars, len;
        bool first = true;
        while ((len = next_token(q + st, i - st, pos, tok, chars)) > 0) {
            // «ВЛ-80» даёт два терма — между ними неявное И
            if (!first) out.push({T_AND, 0});
            if (idx.stemmed) {                     // тот же стеммер, что при индексации
                int sl = stem::stem(tok, len, sb);
                out.push({T_TERM, idx.find_term(sb, sl)});
            } else {
                out.push({T_TERM, idx.find_term(tok, len)});
            }
            first = false;
        }
    }
}

Парсер — рекурсивный спуск#

Грамматика с приоритетом НЕ > И > ИЛИ:

expr  := conj ( '||' conj )*
conj  := unary ( ['&&'] unary )*        — пробел между операндами = И
unary := '!' unary | '(' expr ')' | TERM

Каждому правилу — функция; результат — дерево выражения в массиве узлов. ноги || красный автомобиль разбирается как ноги || (красный && автомобиль).

Толерантность: повторные && пропускаются; оператор без операнда игнорируется; незакрытая ( закрывается в конце; лишняя ) пропускается, а хвост после неё склеивается через И; пустой запрос — пустая выдача.

lab_boolean/src/search.cpp — Parser строки 158–205
struct Parser {
    const Vec<QTok> &t;
    size_t p = 0;
    Vec<Node> nodes;
    explicit Parser(const Vec<QTok> &toks) : t(toks) {}

    int make(NodeType ty, int l, int r, long e = 0) { nodes.push({ty, l, r, e}); return (int)nodes.n - 1; }
    bool at(TokType ty) const { return p < t.n && t[p].type == ty; }

    int unary() {
        if (at(T_NOT)) { p++; int c = unary(); return c < 0 ? -1 : make(N_NOT, c, -1); }
        if (at(T_LP)) { p++; int e = expr(); if (at(T_RP)) p++; return e; }
        if (at(T_TERM)) { long e = t[p].entry; p++; return make(N_TERM, -1, -1, e); }
        return -1;
    }
    int conj() {
        int left = unary();
        for (;;) {
            if (at(T_AND)) { p++; continue; }               // «&&» и повторные «&&» просто разделители
            if (at(T_TERM) || at(T_NOT) || at(T_LP)) {
                int right = unary();
                if (right < 0) continue;
                left = left < 0 ? right : make(N_AND, left, right);
                continue;
            }
            return left;
        }
    }
    int expr() {
        int left = conj();
        while (at(T_OR)) {
            p++;
            int right = conj();
            if (right < 0) continue;
            left = left < 0 ? right : make(N_OR, left, right);
        }
        return left;
    }
    int parse() {
        int root = expr();
        while (p < t.n) {                                   // хвост после лишней «)» — склеиваем через И
            p++;
            int more = expr();
            if (more >= 0) root = root < 0 ? more : make(N_AND, root, more);
        }
        return root;
    }
};

Вычисление дерева#

Рекурсивно снизу вверх; узел A && !B распознаётся и считается вычитанием:

lab_boolean/src/search.cpp — eval строки 207–222
// ---------------------------------------------------------------- вычисление
static void eval(const Index &idx, const Vec<Node> &nodes, int v, Vec<uint32_t> &res) {
    const Node &nd = nodes[v];
    if (nd.type == N_TERM) { idx.postings_of(nd.entry, res); return; }
    Vec<uint32_t> a, b;
    if (nd.type == N_NOT) { eval(idx, nodes, nd.l, a); op_not(a, idx.nDocs, res); a.release(); return; }
    if (nd.type == N_AND && nodes[nd.r].type == N_NOT) {    // A && !B — вычитание без построения дополнения
        eval(idx, nodes, nd.l, a); eval(idx, nodes, nodes[nd.r].l, b); op_and_not(a, b, res);
    } else if (nd.type == N_AND && nodes[nd.l].type == N_NOT) {
        eval(idx, nodes, nd.r, a); eval(idx, nodes, nodes[nd.l].l, b); op_and_not(a, b, res);
    } else {
        eval(idx, nodes, nd.l, a); eval(idx, nodes, nd.r, b);
        if (nd.type == N_AND) op_and(a, b, res); else op_or(a, b, res);
    }
    a.release(); b.release();
}

4. Утилита командной строки search.exe#

search index.bin [-n 50]            — каждая строка stdin = запрос, результат в stdout
search index.bin --paged [-n 50]    — строка = «смещение<TAB>запрос» (для веб-сервиса)
search index.bin --ids              — номера документов вместо заголовков (для тестов)

Формат вывода: QUERY <запрос>, TOTAL <найдено> TIME_US <мкс>, строки «номер \t заголовок \t URL», END. Индекс загружается один раз, дальше запросы обрабатываются в цикле.

lab_boolean/src/search.cpp — main строки 224–276
int main(int argc, char **argv) {
    if (argc < 2) { fprintf(stderr, "usage: search index.bin [--paged] [-n 50]\n"); return 1; }
    bool paged = false, idsOnly = false; long limit = 50;
    for (int i = 2; i < argc; i++) {
        if (!strcmp(argv[i], "--paged")) paged = true;
        else if (!strcmp(argv[i], "--ids")) idsOnly = true;
        else if (!strcmp(argv[i], "-n") && i + 1 < argc) limit = atol(argv[++i]);
    }
    Index idx;
    double tl = now_sec();
    if (!idx.load(argv[1])) return 1;
    fprintf(stderr, "index loaded: %u docs, %u terms%s, %.2f s\n", idx.nDocs, idx.nTerms,
            idx.stemmed ? " (stemmed)" : "", now_sec() - tl);

    Vec<char> line; Vec<QTok> toks; Vec<uint32_t> res;
    int ch;
    for (;;) {
        line.clear();
        while ((ch = getchar()) != EOF && ch != '\n') line.push((char)ch);
        if (ch == EOF && line.n == 0) break;
        if (line.n && line[line.n - 1] == '\r') line.n--;
        const char *q = line.a; size_t qn = line.n;
        long offset = 0;
        if (paged) {
            size_t i = 0;
            while (i < qn && q[i] != '\t') i++;
            offset = atol(q);               // число до табуляции
            if (i < qn) { q += i + 1; qn -= i + 1; }
        }

        double t0 = now_sec();
        lex(idx, q, qn, toks);
        Parser ps(toks);
        int root = ps.parse();
        if (root >= 0) eval(idx, ps.nodes, root, res); else res.clear();
        double us = (now_sec() - t0) * 1e6;

        fputs("QUERY\t", stdout); fwrite(q, 1, qn, stdout); putchar('\n');
        printf("TOTAL\t%zu\tTIME_US\t%.0f\n", res.n, us);
        for (long k = offset; k < (long)res.n && k < offset + limit; k++) {
            if (idsOnly) { printf("%u\n", res[k]); continue; }
            const char *title, *url; int tlen, ulen;
            idx.doc_info(res[k], title, tlen, url, ulen);
            printf("%ld\t", k + 1); fwrite(title, 1, tlen, stdout); putchar('\t');
            fwrite(url, 1, ulen, stdout); putchar('\n');
        }
        puts("END");
        fflush(stdout);
        ps.nodes.release();
        if (ch == EOF) break;
    }
    return 0;
}

5. Веб-сервис#

web/server.py (Python, http.server): стартовая страница с формой; страница выдачи — 50 результатов и ссылка «Следующие 50 результатов» (offset). Поиск выполняет тот же search.exe, запущенный один раз как процесс-сервер (общение через stdin/stdout) — индекс загружается однажды. Режим со стеммингом: python web/server.py 8080 stem.

lab_boolean/web/server.py (117 строк)
"""Веб-сервис булева поиска.

Сам поиск выполняет bin/search.exe (C++): сервер один раз запускает его в режиме --paged,
индекс загружается в память один раз, дальше запросы передаются через stdin/stdout.

Страницы:
  /                        — форма ввода запроса
  /search?q=...&page=N     — выдача: форма + 50 результатов (заголовок + ссылка) + ссылка на следующие 50

Запуск: python web/server.py [порт] [stem]   (по умолчанию 8080), затем открыть http://localhost:8080
        stem — искать по индексу со стеммингом (data/index_stem.bin): без учёта словоформ
"""
import html, os, subprocess, sys, threading, urllib.parse
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer

ROOT = os.path.join(os.path.dirname(os.path.abspath(__file__)), '..')
SEARCH = os.path.join(ROOT, 'bin', 'search.exe')
STEM = len(sys.argv) > 2 and sys.argv[2] == 'stem'
INDEX = os.path.join(ROOT, 'data', 'index_stem.bin' if STEM else 'index.bin')
MODE = ' (со стеммингом: без учёта словоформ)' if STEM else ''
PAGE = 50


class Engine:
    """Обёртка над процессом search.exe --paged."""

    def __init__(self):
        self.p = subprocess.Popen([SEARCH, INDEX, '--paged', '-n', str(PAGE)], stdin=subprocess.PIPE,
                                  stdout=subprocess.PIPE, encoding='utf-8', bufsize=1)
        self.lock = threading.Lock()

    def query(self, q, offset):
        q = q.replace('\t', ' ').replace('\n', ' ').replace('\r', ' ')
        with self.lock:
            self.p.stdin.write(f'{offset}\t{q}\n'); self.p.stdin.flush()
            self.p.stdout.readline()                                   # QUERY ...
            _, total, _, us = self.p.stdout.readline().rstrip('\n').split('\t')
            rows = []
            while (line := self.p.stdout.readline().rstrip('\n')) != 'END':
                rank, title, url = line.split('\t')
                rows.append((int(rank), title, url))
        return int(total), float(us), rows

</style>"""


def form(q=''):
    return (f'<form action="/search" method="get"><input type="text" name="q" value="{html.escape(q)}" '
            f'autofocus placeholder="например: (электровоз || тепловоз) колея !узкоколейная">'
            f'<button type="submit">Найти</button></form>')


def page(title, body):
    return (f'<!doctype html><html lang="ru"><head><meta charset="utf-8"><title>{html.escape(title)}</title>'

class Handler(BaseHTTPRequestHandler):
    def send(self, code, text):
        data = text.encode('utf-8')
        self.send_response(code)
        self.send_header('Content-Type', 'text/html; charset=utf-8')
        self.send_header('Content-Length', str(len(data)))
        self.end_headers()
        self.wfile.write(data)

    def do_GET(self):
        u = urllib.parse.urlparse(self.path)
        args = urllib.parse.parse_qs(u.query)
        if u.path == '/':
            body = ('<h1>Булев поиск: железнодорожный транспорт</h1>' + f'<div class="meta">{MODE}</div>' + form() +
                    '<div class="help">Синтаксис: пробел или <b>&amp;&amp;</b> — И, <b>||</b> — ИЛИ, '
                    '<b>!</b> — НЕ, скобки.<br>Примеры: <i>транссибирская магистраль</i>, '
                    '<i>(электровоз || тепловоз) колея</i>, <i>паровоз !англия</i>, '
                    '<i>(maglev || monorail) &amp;&amp; japan</i></div>')
            self.send(200, page('Булев поиск', body))
        elif u.path == '/search':
            q = args.get('q', [''])[0]
            try:
                pg = max(1, int(args.get('page', ['1'])[0]))
            except ValueError:
                pg = 1
            total, us, rows = ENGINE.query(q, (pg - 1) * PAGE)
            items = ''.join(f'<li value="{r}"><a class="t" href="{html.escape(url)}">{html.escape(t)}</a>'

            first = (pg - 1) * PAGE + 1
            meta = (f'Найдено документов: {total}. Показаны {first}–{first + len(rows) - 1}. '
                    f'Время поиска: {us / 1000:.2f} мс.' if rows else f'Найдено документов: {total}.')
            nxt = ''
            if pg * PAGE < total:
                link = '/search?' + urllib.parse.urlencode({'q': q, 'page': pg + 1})
                nxt = f'<a class="next" href="{html.escape(link)}">Следующие {PAGE} результатов →</a>'
            body = (f'<h1><a href="/">Булев поиск</a></h1>{form(q)}<div class="meta">{meta}{MODE}</div>'
                    f'<ol>{items}</ol>{nxt}')

        else:
            self.send(404, page('404', '<h1>Страница не найдена</h1><a href="/">На главную</a>'))

    def log_message(self, fmt, *a):
        pass


if __name__ == '__main__':

7. Как проверялась корректность#

  1. Мини-корпус из 6 документов с заранее известными ответами — каждый оператор, приоритет, скобки, толерантность парсера, регистр, ё/е, ударение, дефис (GoogleTest Lab8_Search).
  2. 9. Вопросы на защите#

  • Как работает пересечение? Два указателя по отсортированным спискам, O(m + n).
  • Приоритет операторов? НЕ > И > ИЛИ; a || b c = a || (b && c).
  • Что такое рекурсивный спуск? Функция на каждое правило грамматики, вызывают друг друга.
  • Как толерантность? Лишние операторы пропускаются, скобки закрываются/игнорируются.
  • Почему !the медленный? Дополнение до 65 тыс. документов.
  • Как доказать, что поиск верный? Сверка с перебором всех документов на 300 случайных запросах.
  • Как веб-сервис получает результаты? Через тот же search.exe (один процесс, stdin/stdout).

// Булев поиск по индексу index.bin.
//
// Синтаксис запросов: пробел или && — И, || — ИЛИ, ! — НЕ, скобки.
// Запуск:
//   search index.bin [-n 50]            — каждая строка stdin = запрос, результат в stdout
//   search index.bin --paged [-n 50]    — каждая строка stdin = "смещение<TAB>запрос" (для веб-сервиса)
//   search index.bin --ids [-n N]       — вместо заголовков печатать внутренние номера документов (для тестов)
#include "common.h"
#include "stem.h"

// ---------------------------------------------------------------- загруженный индекс
struct Index {
    char *data = nullptr;
    uint64_t size = 0;
    const char *docs = nullptr, *dict = nullptr, *termstr = nullptr, *postings = nullptr;
    uint32_t nDocs = 0, nTerms = 0;
    bool stemmed = false;          // в индексе основы слов -> слова запроса тоже стеммируются

    bool load(const char *path) {
        FILE *f = fopen(path, "rb");
        if (!f) { fprintf(stderr, "cannot open %s\n", path); return false; }
        #ifdef _WIN32
        _fseeki64(f, 0, SEEK_END); size = (uint64_t)_ftelli64(f); _fseeki64(f, 0, SEEK_SET);
#else
        fseeko(f, 0, SEEK_END); size = (uint64_t)ftello(f); fseeko(f, 0, SEEK_SET);
#endif
        data = (char *)malloc(size);
        if (fread(data, 1, size, f) != size) { fprintf(stderr, "read error\n"); return false; }
        fclose(f);
        if (memcmp(data, INDEX_MAGIC, 4) != 0) { fprintf(stderr, "bad magic\n"); return false; }
        if (rd32(data + 4) != INDEX_VERSION) { fprintf(stderr, "unsupported version\n"); return false; }
        uint32_t nsec = rd32(data + 8);
        for (uint32_t i = 0; i < nsec; i++) {
            const char *e = data + HEADER_SIZE + i * SECTION_ENTRY_SIZE;
            uint32_t type = rd32(e);
            const char *p = data + rd64(e + 8);
            if (type == SEC_DOCS) docs = p;
            else if (type == SEC_DICT) dict = p;
            else if (type == SEC_TERMSTR) termstr = p;
            else if (type == SEC_POSTINGS) postings = p;
            // неизвестные секции пропускаются — так формат расширяется без поломки старых программ
        }
        if (!docs || !dict || !termstr || !postings) { fprintf(stderr, "missing section\n"); return false; }
        nDocs = rd32(docs);
        nTerms = rd32(dict);
        stemmed = (rd32(data + 12) & FLAG_STEMMED) != 0;   // флаги заголовка
        return true;
    }

    // Двоичный поиск терма в отсортированном словаре. Возвращает номер записи или -1.
    long find_term(const char *s, int n) const {
        long lo = 0, hi = (long)nTerms - 1;
        while (lo <= hi) {
            long mid = (lo + hi) / 2;
            const char *e = dict + 4 + (size_t)mid * DICT_ENTRY_SIZE;
            int c = cmp_bytes(termstr + rd32(e), rd16(e + 4), s, n);
            if (c == 0) return mid;
            if (c < 0) lo = mid + 1; else hi = mid - 1;
        }
        return -1;
    }

    void postings_of(long entry, Vec<uint32_t> &out) const {
        out.clear();
        if (entry < 0) return;
        const char *e = dict + 4 + (size_t)entry * DICT_ENTRY_SIZE;
        uint32_t df = rd32(e + 8);
        out.resize(df);
        memcpy(out.a, postings + rd64(e + 16), (size_t)df * 4);
    }

    // Заголовок и URL документа из прямого индекса.
    void doc_info(uint32_t d, const char *&title, int &tl, const char *&url, int &ul) const {
        const char *r = docs + rd64(docs + 4 + 8ull * d);
        tl = rd16(r + 5); title = r + 7;
        ul = rd16(r + 7 + tl); url = r + 9 + tl;
    }
};

// ---------------------------------------------------------------- операции над списками
static void op_and(const Vec<uint32_t> &a, const Vec<uint32_t> &b, Vec<uint32_t> &r) {
    r.clear(); size_t i = 0, j = 0;
    while (i < a.n && j < b.n) {
        if (a[i] < b[j]) i++;
        else if (a[i] > b[j]) j++;
        else { r.push(a[i]); i++; j++; }
    }
}
static void op_or(const Vec<uint32_t> &a, const Vec<uint32_t> &b, Vec<uint32_t> &r) {
    r.clear(); r.reserve(a.n + b.n); size_t i = 0, j = 0;
    while (i < a.n && j < b.n) {
        if (a[i] < b[j]) r.push(a[i++]);
        else if (a[i] > b[j]) r.push(b[j++]);
        else { r.push(a[i]); i++; j++; }
    }
    while (i < a.n) r.push(a[i++]);
    while (j < b.n) r.push(b[j++]);
}
static void op_and_not(const Vec<uint32_t> &a, const Vec<uint32_t> &b, Vec<uint32_t> &r) {
    r.clear(); size_t j = 0;
    for (size_t i = 0; i < a.n; i++) {
        while (j < b.n && b[j] < a[i]) j++;
        if (j < b.n && b[j] == a[i]) continue;
        r.push(a[i]);
    }
}
static void op_not(const Vec<uint32_t> &a, uint32_t nDocs, Vec<uint32_t> &r) {
    r.clear(); size_t j = 0;
    for (uint32_t d = 0; d < nDocs; d++) {
        if (j < a.n && a[j] == d) { j++; continue; }
        r.push(d);
    }
}

// ---------------------------------------------------------------- лексер запроса
enum TokType { T_TERM, T_AND, T_OR, T_NOT, T_LP, T_RP };
struct QTok { TokType type; long entry; };   // entry — номер терма в словаре (-1, если терма нет)

static void lex(const Index &idx, const char *q, size_t n, Vec<QTok> &out) {
    out.clear();
    size_t i = 0;
    char tok[MAX_TERM_BYTES + 4], sb[MAX_TERM_BYTES + 8];
    while (i < n) {
        char c = q[i];
        if (c == ' ' || c == '\t' || c == '\r' || c == '\n') { i++; continue; }
        if (c == '(') { out.push({T_LP, 0}); i++; continue; }
        if (c == ')') { out.push({T_RP, 0}); i++; continue; }
        if (c == '!') { out.push({T_NOT, 0}); i++; continue; }
        if (c == '&') { while (i < n && q[i] == '&') i++; out.push({T_AND, 0}); continue; }
        if (c == '|') { while (i < n && q[i] == '|') i++; out.push({T_OR, 0}); continue; }
        // слово: всё до пробела или спецсимвола; внутри него работает тот же токенизатор, что и в индексаторе
        size_t st = i;
        while (i < n && !strchr(" \t\r\n()!&|", q[i])) i++;
        size_t pos = 0; int chars, len;
        bool first = true;
        while ((len = next_token(q + st, i - st, pos, tok, chars)) > 0) {
            // «ВЛ-80» даёт два терма — между ними неявное И
            if (!first) out.push({T_AND, 0});
            if (idx.stemmed) {                     // тот же стеммер, что при индексации
                int sl = stem::stem(tok, len, sb);
                out.push({T_TERM, idx.find_term(sb, sl)});
            } else {
                out.push({T_TERM, idx.find_term(tok, len)});
            }
            first = false;
        }
    }
}

// ---------------------------------------------------------------- парсер (рекурсивный спуск)
// or  := and ( '||' and )*
// and := unary ( ['&&'] unary )*        — пробел между операндами = И
// unary := '!' unary | '(' or ')' | TERM
// Парсер толерантен: лишние операторы и скобки пропускаются, незакрытые скобки закрываются.
enum NodeType { N_TERM, N_AND, N_OR, N_NOT };
struct Node { NodeType type; int l, r; long entry; };

struct Parser {
    const Vec<QTok> &t;
    size_t p = 0;
    Vec<Node> nodes;
    explicit Parser(const Vec<QTok> &toks) : t(toks) {}

    int make(NodeType ty, int l, int r, long e = 0) { nodes.push({ty, l, r, e}); return (int)nodes.n - 1; }
    bool at(TokType ty) const { return p < t.n && t[p].type == ty; }

    int unary() {
        if (at(T_NOT)) { p++; int c = unary(); return c < 0 ? -1 : make(N_NOT, c, -1); }
        if (at(T_LP)) { p++; int e = expr(); if (at(T_RP)) p++; return e; }
        if (at(T_TERM)) { long e = t[p].entry; p++; return make(N_TERM, -1, -1, e); }
        return -1;
    }
    int conj() {
        int left = unary();
        for (;;) {
            if (at(T_AND)) { p++; continue; }               // «&&» и повторные «&&» просто разделители
            if (at(T_TERM) || at(T_NOT) || at(T_LP)) {
                int right = unary();
                if (right < 0) continue;
                left = left < 0 ? right : make(N_AND, left, right);
                continue;
            }
            return left;
        }
    }
    int expr() {
        int left = conj();
        while (at(T_OR)) {
            p++;
            int right = conj();
            if (right < 0) continue;
            left = left < 0 ? right : make(N_OR, left, right);
        }
        return left;
    }
    int parse() {
        int root = expr();
        while (p < t.n) {                                   // хвост после лишней «)» — склеиваем через И
            p++;
            int more = expr();
            if (more >= 0) root = root < 0 ? more : make(N_AND, root, more);
        }
        return root;
    }
};

// ---------------------------------------------------------------- вычисление
static void eval(const Index &idx, const Vec<Node> &nodes, int v, Vec<uint32_t> &res) {
    const Node &nd = nodes[v];
    if (nd.type == N_TERM) { idx.postings_of(nd.entry, res); return; }
    Vec<uint32_t> a, b;
    if (nd.type == N_NOT) { eval(idx, nodes, nd.l, a); op_not(a, idx.nDocs, res); a.release(); return; }
    if (nd.type == N_AND && nodes[nd.r].type == N_NOT) {    // A && !B — вычитание без построения дополнения
        eval(idx, nodes, nd.l, a); eval(idx, nodes, nodes[nd.r].l, b); op_and_not(a, b, res);
    } else if (nd.type == N_AND && nodes[nd.l].type == N_NOT) {
        eval(idx, nodes, nd.r, a); eval(idx, nodes, nodes[nd.l].l, b); op_and_not(a, b, res);
    } else {
        eval(idx, nodes, nd.l, a); eval(idx, nodes, nd.r, b);
        if (nd.type == N_AND) op_and(a, b, res); else op_or(a, b, res);
    }
    a.release(); b.release();
}

int main(int argc, char **argv) {
    if (argc < 2) { fprintf(stderr, "usage: search index.bin [--paged] [-n 50]\n"); return 1; }
    bool paged = false, idsOnly = false; long limit = 50;
    for (int i = 2; i < argc; i++) {
        if (!strcmp(argv[i], "--paged")) paged = true;
        else if (!strcmp(argv[i], "--ids")) idsOnly = true;
        else if (!strcmp(argv[i], "-n") && i + 1 < argc) limit = atol(argv[++i]);
    }
    Index idx;
    double tl = now_sec();
    if (!idx.load(argv[1])) return 1;
    fprintf(stderr, "index loaded: %u docs, %u terms%s, %.2f s\n", idx.nDocs, idx.nTerms,
            idx.stemmed ? " (stemmed)" : "", now_sec() - tl);

    Vec<char> line; Vec<QTok> toks; Vec<uint32_t> res;
    int ch;
    for (;;) {
        line.clear();
        while ((ch = getchar()) != EOF && ch != '\n') line.push((char)ch);
        if (ch == EOF && line.n == 0) break;
        if (line.n && line[line.n - 1] == '\r') line.n--;
        const char *q = line.a; size_t qn = line.n;
        long offset = 0;
        if (paged) {
            size_t i = 0;
            while (i < qn && q[i] != '\t') i++;
            offset = atol(q);               // число до табуляции
            if (i < qn) { q += i + 1; qn -= i + 1; }
        }

        double t0 = now_sec();
        lex(idx, q, qn, toks);
        Parser ps(toks);
        int root = ps.parse();
        if (root >= 0) eval(idx, ps.nodes, root, res); else res.clear();
        double us = (now_sec() - t0) * 1e6;

        fputs("QUERY\t", stdout); fwrite(q, 1, qn, stdout); putchar('\n');
        printf("TOTAL\t%zu\tTIME_US\t%.0f\n", res.n, us);
        for (long k = offset; k < (long)res.n && k < offset + limit; k++) {
            if (idsOnly) { printf("%u\n", res[k]); continue; }
            const char *title, *url; int tlen, ulen;
            idx.doc_info(res[k], title, tlen, url, ulen);
            printf("%ld\t", k + 1); fwrite(title, 1, tlen, stdout); putchar('\t');
            fwrite(url, 1, ulen, stdout); putchar('\n');
        }
        puts("END");
        fflush(stdout);
        ps.nodes.release();
        if (ch == EOF) break;
    }
    return 0;
}

    pos = [0]

    def at(*v):
        return pos[0] < len(toks) and names.get(toks[pos[0]], 'TERM') in v

    def unary():
        if at('NOT'):
            pos[0] += 1; c = unary(); return ('НЕ', c) if c else None
        if at('LP'):
            pos[0] += 1; e = expr()
            if at('RP'): pos[0] += 1
            return e
        if at('TERM'):
            t = toks[pos[0]]; pos[0] += 1; return t
        return None

    def conj():
        left = unary()
        while True:
            if at('AND'):
                pos[0] += 1; continue
            if at('TERM', 'NOT', 'LP'):
                r = unary()
                if r: left = ('И', left, r) if left else r
                continue
            return left

    def expr():
        left = conj()
        while at('OR'):
            pos[0] += 1; r = conj()
            if r: left = ('ИЛИ', left, r) if left else r
        return left

    tree = expr()

    def draw(n, ind=''):
        if isinstance(n, tuple):
            print(ind + n[0]); [draw(c, ind + '   ') for c in n[1:]]
        else:
            print(ind + str(n))
    print('дерево (приоритет: НЕ > И > ИЛИ):')
    draw(tree, '  ')
    total, us, _ = search(q, n=1)
    print(f'результат: {fmt(total)} документов, {us} мкс')


def cmd_bench(_):
