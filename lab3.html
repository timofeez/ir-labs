ЛР3. Токенизация#

Задание (lesson-04/homework.md): реализовать разбиение текстов документов на токены, которое используется при индексации. Описать правила, достоинства и недостатки. Привести примеры неудачно выделенных токенов и объяснить, как поправить правила. Указать число токенов и среднюю длину токена, время работы, зависимость времени от объёма данных, скорость на килобайт текста; оптимальна ли скорость и как её ускорить.

1. Токен, терм, нормализация#

  • Токен — конкретное вхождение слова в тексте. В «поезд и поезда» три токена.
  • Терм — нормализованный токен, запись в словаре индекса: «Поезд», «ПОЕЗД», «поезд» → терм поезд.
  • Токенизация — разбиение текста на токены. Первый шаг любой обработки: индекс, поиск, статистика работают со словами, а не с сырым текстом.
  • Нормализация — приведение токенов к одному виду: регистр, «ё» → «е», ударения, диакритика.

Главное правило: документы и запросы должны токенизироваться одной и той же функцией. Если индексатор режет «ВЛ-80» на вл + 80, а поиск искал бы вл-80, слово никогда бы не нашлось. У меня это функция next_token из common.h — её вызывают индексатор (ЛР7), поиск (ЛР8), подсчёт частот (ЛР5) и утилита ЛР3.

2. Мои правила#

  1. Символ слова — буква или цифра: латиница a–z A–Z, цифры, латиница с диакритикой (U+00C0–U+024F, кроме × и ÷), греческий (U+0370–U+03FF), кириллица (U+0400–U+04FF).
  2. Комбинируемые знаки U+0300–U+036F (ударение «´» и т. п.) пропускаются — слово не рвётся: «Железнодоро́жный» → железнодорожный.
  3. Всё остальное — разделитель: пробелы, пунктуация, дефис, апостроф, @, иероглифы.
  4. Нижний регистр для латиницы, кириллицы, греческого; ё → е.
  5. Длина токена — не больше 64 байт (защита от «слов» из мусора).

Это тот же подход, что у токенизатора LetterTokenizer + LowerCaseFilter в Lucene (основа Elasticsearch) — базовый уровень; следующий уровень — правила Unicode для границ слов (UAX #29, StandardTokenizer).

3. UTF-8#

Тексты в UTF-8: символ занимает 1–4 байта. По старшим битам первого байта видно длину символа, байты продолжения начинаются с 10:

Байт Значит Пример
0xxxxxxx 1 байт (ASCII) A = 41
110xxxxx 10xxxxxx 2 байта Ж = D0 96 (U+0416)
1110xxxx 10xxxxxx 10xxxxxx 3 байта € = E2 82 AC
11110xxx … 4 байта эмодзи

Декодирование: из первого байта берутся младшие биты, из каждого следующего — младшие 6 бит (& 0x3F), всё склеивается сдвигом на 6. Испорченный байт превращается в U+FFFD, программа не падает:

lab_boolean/src/common.h — utf8_decode строки 42–53
// ---------------------------------------------------------------- UTF-8 и токенизация
// Декодирует один символ UTF-8 начиная с s[i], сдвигает i.
static inline uint32_t utf8_decode(const unsigned char *s, size_t n, size_t &i) {
    unsigned c = s[i];
    if (c < 0x80) { i++; return c; }
    int len = c >= 0xF0 ? 4 : c >= 0xE0 ? 3 : c >= 0xC0 ? 2 : 1;
    if (len == 1 || i + len > n) { i++; return 0xFFFD; }
    uint32_t cp = c & (0xFF >> (len + 1));
    for (int k = 1; k < len; k++) cp = (cp << 6) | (s[i + k] & 0x3F);
    i += len;
    return cp;
}
lab_boolean/src/common.h — utf8_encode строки 55–64
static inline int utf8_encode(uint32_t c, char *o) {
    if (c < 0x80) { o[0] = (char)c; return 1; }
    if (c < 0x800) { o[0] = (char)(0xC0 | (c >> 6)); o[1] = (char)(0x80 | (c & 0x3F)); return 2; }
    if (c < 0x10000) {
        o[0] = (char)(0xE0 | (c >> 12)); o[1] = (char)(0x80 | ((c >> 6) & 0x3F));
        o[2] = (char)(0x80 | (c & 0x3F)); return 3;
    }
    o[0] = (char)(0xF0 | (c >> 18)); o[1] = (char)(0x80 | ((c >> 12) & 0x3F));
    o[2] = (char)(0x80 | ((c >> 6) & 0x3F)); o[3] = (char)(0x80 | (c & 0x3F)); return 4;
}

4. Код токенизатора#

Класс символа и нормализация регистра — таблицами диапазонов Unicode:

lab_boolean/src/common.h — char_class строки 66–74
// Класс символа: 1 — часть слова, 2 — игнорируемый (знак ударения), 0 — разделитель.
static inline int char_class(uint32_t c) {
    if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9')) return 1;
    if (c >= 0x300 && c <= 0x36F) return 2;                 // комбинируемые диакритики (ударения)
    if (c >= 0xC0 && c <= 0x24F && c != 0xD7 && c != 0xF7) return 1;  // латиница с диакритикой
    if (c >= 0x370 && c <= 0x3FF) return 1;                 // греческий
    if (c >= 0x400 && c <= 0x4FF) return 1;                 // кириллица
    return 0;
}
lab_boolean/src/common.h — to_lower строки 76–85
// Понижение регистра + нормализация «ё» -> «е».
static inline uint32_t to_lower(uint32_t c) {
    if (c >= 'A' && c <= 'Z') return c + 32;
    if (c == 0x401 || c == 0x451) return 0x435;             // Ё, ё -> е
    if (c >= 0x410 && c <= 0x42F) return c + 0x20;          // А-Я
    if (c >= 0x400 && c <= 0x40F) return c + 0x50;          // Ѐ-Џ
    if (c >= 0xC0 && c <= 0xDE && c != 0xD7) return c + 0x20;
    if (c >= 0x391 && c <= 0x3AB) return c + 0x20;          // греческие заглавные
    return c;
}

Сама функция: идёт по тексту символ за символом; буква — дописывается к токену (в нижнем регистре), ударение — пропускается, разделитель после начатого слова — конец токена. pos передаётся по ссылке, поэтому функция «помнит», где остановилась, и вызывается в цикле while ((len = next_token(...)) > 0):

lab_boolean/src/common.h — next_token строки 89–109
// Читает следующий токен из s[0..n) начиная с pos.
// В out пишется нормализованный терм (UTF-8, нижний регистр), возвращается его длина в байтах
// (0 — токенов больше нет). В chars — длина в символах.
static inline int next_token(const char *s, size_t n, size_t &pos, char *out, int &chars) {
    const unsigned char *u = (const unsigned char *)s;
    int len = 0; chars = 0;
    while (pos < n) {
        size_t start = pos;
        uint32_t c = utf8_decode(u, n, pos);
        int cls = char_class(c);
        if (cls == 1) {
            if (len + 4 <= MAX_TERM_BYTES) { len += utf8_encode(to_lower(c), out + len); chars++; }
        } else if (cls == 0) {
            if (len > 0) { pos = start; break; }
        }
        // cls == 2: знак ударения внутри слова просто пропускается
    }
    // разделитель, на котором остановились, пропустим при следующем вызове
    if (len > 0 && pos < n) { size_t p = pos; utf8_decode(u, n, p); pos = p; }
    return len;
}

5. Утилита командной строки tokenize.exe#

Два режима: tokenize < текст — токены по одному на строку; tokenize --stats corpus.tsv [--limit-mb N] — статистика по корпусу (время засекается отдельно для токенизации, без чтения файла):

lab_boolean/src/tokenize.cpp — stats строки 23–63
static int stats(const char *path, double limitMb) {
    FILE *in = fopen(path, "rb");
    if (!in) { fprintf(stderr, "cannot open %s\n", path); return 1; }
    LineReader lr(in);
    uint64_t limit = limitMb > 0 ? (uint64_t)(limitMb * 1024 * 1024) : UINT64_MAX;
    uint64_t textBytes = 0, tokens = 0, tokenChars = 0, tokenBytes = 0, docs = 0;
    double tokenizeSec = 0, t0 = now_sec();
    char tok[MAX_TERM_BYTES + 4];
    while (textBytes < limit && lr.next()) {
        char *s = lr.line.a; size_t n = lr.line.n;
        if (n && s[n - 1] == '\r') n--;
        // поля: источник \t id \t заголовок \t URL \t текст
        char *f[5]; size_t fl[5]; int k = 0; size_t st = 0;
        for (size_t i = 0; i <= n && k < 5; i++)
            if (i == n || (s[i] == '\t' && k < 4)) { f[k] = s + st; fl[k] = i - st; k++; st = i + 1; }
        if (k < 5) continue;
        docs++;
        double ts = now_sec();
        for (int part = 2; part <= 4; part += 2) {
            size_t pos = 0; int chars, len;
            textBytes += fl[part];
            while ((len = next_token(f[part], fl[part], pos, tok, chars)) > 0) {
                tokens++; tokenChars += chars; tokenBytes += len;
            }
        }
        tokenizeSec += now_sec() - ts;
    }
    fclose(in);
    double total = now_sec() - t0;
    double kb = textBytes / 1024.0;
    printf("documents          %llu\n", (unsigned long long)docs);
    printf("text_bytes         %llu\n", (unsigned long long)textBytes);
    printf("tokens             %llu\n", (unsigned long long)tokens);
    printf("avg_token_len      %.3f\n", tokens ? (double)tokenChars / tokens : 0.0);   // в символах
    printf("avg_token_bytes    %.3f\n", tokens ? (double)tokenBytes / tokens : 0.0);   // в байтах UTF-8
    printf("time_total_sec     %.3f\n", total);          // чтение файла + токенизация
    printf("time_tokenize_sec  %.3f\n", tokenizeSec);    // только токенизация
    printf("ms_per_kb          %.5f\n", kb ? tokenizeSec * 1000 / kb : 0.0);
    printf("mb_per_sec         %.2f\n", tokenizeSec ? textBytes / 1048576.0 / tokenizeSec : 0.0);
    return 0;
}

lab_boolean/src/common.h — Vec строки 13–33
struct Vec {
    T *a = nullptr;
    size_t n = 0, cap = 0;

    void grow(size_t c) {
        T *b = (T *)realloc(a, c * sizeof(T));
        if (!b) { fprintf(stderr, "out of memory\n"); exit(1); }
        a = b; cap = c;
    }
    void reserve(size_t c) { if (c > cap) grow(c); }
    void push(const T &x) { if (n == cap) grow(cap ? cap * 2 : 16); a[n++] = x; }
    void append(const T *src, size_t k) {
        if (n + k > cap) { size_t c = cap ? cap : 16; while (c < n + k) c *= 2; grow(c); }
        memcpy(a + n, src, k * sizeof(T)); n += k;
    }
    void resize(size_t k) { reserve(k); n = k; }
    void clear() { n = 0; }
    void release() { free(a); a = nullptr; n = cap = 0; }
    T &operator[](size_t i) { return a[i]; }
    const T &operator[](size_t i) const { return a[i]; }
};
lab_boolean/src/common.h — LineReader строки 137–163
// ---------------------------------------------------------------- построчное чтение файла
struct LineReader {
    FILE *f;
    char *buf;
    size_t cap = 1 << 20, beg = 0, end = 0;
    bool eof = false;
    Vec<char> line;

    explicit LineReader(FILE *file) : f(file) { buf = (char *)malloc(cap); }
    ~LineReader() { free(buf); line.release(); }

    bool next() {
        line.clear();
        for (;;) {
            if (beg == end) {
                if (eof) return line.n > 0;
                end = fread(buf, 1, cap, f); beg = 0;
                if (end == 0) { eof = true; return line.n > 0; }
            }
            char *p = (char *)memchr(buf + beg, '\n', end - beg);
            size_t stop = p ? (size_t)(p - buf) : end;
            line.append(buf + beg, stop - beg);
            beg = p ? stop + 1 : end;
            if (p) return true;
        }
    }
};

6. Результаты#

Показатель Значение
Документов 65 509
Объём текста (заголовки + тексты) 315 МБ
Токенов 36 011 927
Средняя длина токена 5,43 символа (7,93 байта: кириллица — 2 байта на букву)
Время токенизации 1,40 с
Скорость 0,0043 мс/КБ, ~225 МБ/с
Зависимость от объёма линейная, R² = 0,999

Почему линейно: каждый байт читается и обрабатывается ровно один раз, сложность O(n), памяти — O(1) кроме буфера. Оптимальна ли скорость? Для учебной задачи — да: токенизация занимает ~1,4 с из ~4,9 с индексации, узкое место — чтение и хеширование термов. Как ускорить: таблица классов на 256 элементов для ASCII вместо цепочки сравнений; разбор байтов без полного декодирования (у кириллицы первый байт D0/D1); многопоточность (части корпуса — разным потокам); отображение файла в память (mmap).

10. Вопросы на защите#

  • Чем токен отличается от терма? Токен — вхождение, терм — нормализованная запись словаря.
  • Почему средняя длина токена (5,43) меньше средней длины терма (8,38)? Токены считаются по вхождениям — среди них много коротких частых слов (и, в, the, of); терм в словаре считается один раз, а словарь в основном из редких длинных слов.
  • Почему 7,93 байта на токен? Кириллица в UTF-8 — 2 байта на букву.
  • Как понять длину символа UTF-8? По старшим битам первого байта.
  • Почему токенизатор такой простой? Это базовый вариант (как LetterTokenizer в Lucene); его недостатки найдены на корпусе и описаны с исправлениями.
  • Где ещё используется? Индексатор, поиск (слова запроса), подсчёт частот — одна функция.
  • Стоп-слова? Не удаляю: нужны для поиска фраз и в булевом поиске не мешают; частые слова видны в ЛР5.

lab_boolean/src/common.h (210 строк)
// Общие структуры данных и функции для индексатора и поиска.
// STL не используется: только стандартная библиотека C (cstdio, cstdlib, cstring).
#pragma once
#include <cstdio>
#include <cstdlib>
#include <cstring>
#include <cstdint>
#include <ctime>

// ---------------------------------------------------------------- динамический массив
// Аналог vector для простых (POD) типов: рост в 2 раза, память через realloc.
template <class T>
struct Vec {
    T *a = nullptr;
    size_t n = 0, cap = 0;

    void grow(size_t c) {
        T *b = (T *)realloc(a, c * sizeof(T));
        if (!b) { fprintf(stderr, "out of memory\n"); exit(1); }
        a = b; cap = c;
    }
    void reserve(size_t c) { if (c > cap) grow(c); }
    void push(const T &x) { if (n == cap) grow(cap ? cap * 2 : 16); a[n++] = x; }
    void append(const T *src, size_t k) {
        if (n + k > cap) { size_t c = cap ? cap : 16; while (c < n + k) c *= 2; grow(c); }
        memcpy(a + n, src, k * sizeof(T)); n += k;
    }
    void resize(size_t k) { reserve(k); n = k; }
    void clear() { n = 0; }
    void release() { free(a); a = nullptr; n = cap = 0; }
    T &operator[](size_t i) { return a[i]; }
    const T &operator[](size_t i) const { return a[i]; }
};

// ---------------------------------------------------------------- время
static inline double now_sec() {
    timespec ts;
    timespec_get(&ts, TIME_UTC);
    return ts.tv_sec + ts.tv_nsec * 1e-9;
}

// ---------------------------------------------------------------- UTF-8 и токенизация
// Декодирует один символ UTF-8 начиная с s[i], сдвигает i.
static inline uint32_t utf8_decode(const unsigned char *s, size_t n, size_t &i) {
    unsigned c = s[i];
    if (c < 0x80) { i++; return c; }
    int len = c >= 0xF0 ? 4 : c >= 0xE0 ? 3 : c >= 0xC0 ? 2 : 1;
    if (len == 1 || i + len > n) { i++; return 0xFFFD; }
    uint32_t cp = c & (0xFF >> (len + 1));
    for (int k = 1; k < len; k++) cp = (cp << 6) | (s[i + k] & 0x3F);
    i += len;
    return cp;
}

static inline int utf8_encode(uint32_t c, char *o) {
    if (c < 0x80) { o[0] = (char)c; return 1; }
    if (c < 0x800) { o[0] = (char)(0xC0 | (c >> 6)); o[1] = (char)(0x80 | (c & 0x3F)); return 2; }
    if (c < 0x10000) {
        o[0] = (char)(0xE0 | (c >> 12)); o[1] = (char)(0x80 | ((c >> 6) & 0x3F));
        o[2] = (char)(0x80 | (c & 0x3F)); return 3;
    }
    o[0] = (char)(0xF0 | (c >> 18)); o[1] = (char)(0x80 | ((c >> 12) & 0x3F));
    o[2] = (char)(0x80 | ((c >> 6) & 0x3F)); o[3] = (char)(0x80 | (c & 0x3F)); return 4;
}

// Класс символа: 1 — часть слова, 2 — игнорируемый (знак ударения), 0 — разделитель.
static inline int char_class(uint32_t c) {
    if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9')) return 1;
    if (c >= 0x300 && c <= 0x36F) return 2;                 // комбинируемые диакритики (ударения)
    if (c >= 0xC0 && c <= 0x24F && c != 0xD7 && c != 0xF7) return 1;  // латиница с диакритикой
    if (c >= 0x370 && c <= 0x3FF) return 1;                 // греческий
    if (c >= 0x400 && c <= 0x4FF) return 1;                 // кириллица
    return 0;
}

// Понижение регистра + нормализация «ё» -> «е».
static inline uint32_t to_lower(uint32_t c) {
    if (c >= 'A' && c <= 'Z') return c + 32;
    if (c == 0x401 || c == 0x451) return 0x435;             // Ё, ё -> е
    if (c >= 0x410 && c <= 0x42F) return c + 0x20;          // А-Я
    if (c >= 0x400 && c <= 0x40F) return c + 0x50;          // Ѐ-Џ
    if (c >= 0xC0 && c <= 0xDE && c != 0xD7) return c + 0x20;
    if (c >= 0x391 && c <= 0x3AB) return c + 0x20;          // греческие заглавные
    return c;
}

const int MAX_TERM_BYTES = 64;

// Читает следующий токен из s[0..n) начиная с pos.
// В out пишется нормализованный терм (UTF-8, нижний регистр), возвращается его длина в байтах
// (0 — токенов больше нет). В chars — длина в символах.
static inline int next_token(const char *s, size_t n, size_t &pos, char *out, int &chars) {
    const unsigned char *u = (const unsigned char *)s;
    int len = 0; chars = 0;
    while (pos < n) {
        size_t start = pos;
        uint32_t c = utf8_decode(u, n, pos);
        int cls = char_class(c);
        if (cls == 1) {
            if (len + 4 <= MAX_TERM_BYTES) { len += utf8_encode(to_lower(c), out + len); chars++; }
        } else if (cls == 0) {
            if (len > 0) { pos = start; break; }
        }
        // cls == 2: знак ударения внутри слова просто пропускается
    }
    // разделитель, на котором остановились, пропустим при следующем вызове
    if (len > 0 && pos < n) { size_t p = pos; utf8_decode(u, n, p); pos = p; }
    return len;
}

struct LineReader {
    FILE *f;
    char *buf;
    size_t cap = 1 << 20, beg = 0, end = 0;
    bool eof = false;
    Vec<char> line;

    explicit LineReader(FILE *file) : f(file) { buf = (char *)malloc(cap); }
    ~LineReader() { free(buf); line.release(); }

    bool next() {
        line.clear();
        for (;;) {
            if (beg == end) {
                if (eof) return line.n > 0;
                end = fread(buf, 1, cap, f); beg = 0;
                if (end == 0) { eof = true; return line.n > 0; }
            }
            char *p = (char *)memchr(buf + beg, '\n', end - beg);
            size_t stop = p ? (size_t)(p - buf) : end;
            line.append(buf + beg, stop - beg);
            beg = p ? stop + 1 : end;
            if (p) return true;
        }
    }
};

// Открытая адресация с линейным пробированием; строки термов хранятся в общем пуле.
struct TermTable {
    struct Slot { uint32_t hash, id; };
    Slot *slots = nullptr;
    size_t cap = 0, size = 0;
    Vec<char> pool;            // байты всех термов подряд
    Vec<uint32_t> off;         // off[id]  — смещение терма в пуле
    Vec<uint8_t> len;          // len[id]  — длина терма в байтах

    static uint32_t fnv1a(const char *s, int n) {
        uint32_t h = 2166136261u;
        for (int i = 0; i < n; i++) { h ^= (unsigned char)s[i]; h *= 16777619u; }
        return h;
    }
    void rehash(size_t newcap) {
        Slot *ns = (Slot *)malloc(newcap * sizeof(Slot));
        for (size_t i = 0; i < newcap; i++) ns[i].id = UINT32_MAX;
        for (size_t i = 0; i < cap; i++) {
            if (slots[i].id == UINT32_MAX) continue;
            size_t j = slots[i].hash & (newcap - 1);
            while (ns[j].id != UINT32_MAX) j = (j + 1) & (newcap - 1);
            ns[j] = slots[i];
        }
        free(slots); slots = ns; cap = newcap;
    }
    // Возвращает id терма; если терма ещё нет — добавляет. isNew = true для нового.
    uint32_t get_or_add(const char *s, int n, bool &isNew) {
        if (size * 2 >= cap) rehash(cap ? cap * 2 : 1 << 16);
        uint32_t h = fnv1a(s, n);
        size_t j = h & (cap - 1);
        while (slots[j].id != UINT32_MAX) {
            uint32_t id = slots[j].id;
            if (slots[j].hash == h && len[id] == n && memcmp(pool.a + off[id], s, n) == 0) {
                isNew = false; return id;
            }
            j = (j + 1) & (cap - 1);
        }
        uint32_t id = (uint32_t)size++;
        slots[j].hash = h; slots[j].id = id;
        off.push((uint32_t)pool.n); len.push((uint8_t)n); pool.append(s, n);
        isNew = true;
        return id;
    }
};

// Токенизатор (ЛР «Токенизация») — утилита командной строки. Правила токенизации — next_token() в common.h,

//                                         — токенизирует заголовки и тексты корпуса (не больше N МБ текста),

#include "common.h"

static void tokens_of_stdin() {
    Vec<char> text;
    char buf[1 << 16];
    size_t k;
    while ((k = fread(buf, 1, sizeof(buf), stdin)) > 0) text.append(buf, k);
    char tok[MAX_TERM_BYTES + 4];
    size_t pos = 0; int chars, len;
    while ((len = next_token(text.a, text.n, pos, tok, chars)) > 0) {
        fwrite(tok, 1, len, stdout);
        putchar('\n');
    }
}

static int stats(const char *path, double limitMb) {
    FILE *in = fopen(path, "rb");
    if (!in) { fprintf(stderr, "cannot open %s\n", path); return 1; }
    LineReader lr(in);
    uint64_t limit = limitMb > 0 ? (uint64_t)(limitMb * 1024 * 1024) : UINT64_MAX;
    uint64_t textBytes = 0, tokens = 0, tokenChars = 0, tokenBytes = 0, docs = 0;
    double tokenizeSec = 0, t0 = now_sec();
    char tok[MAX_TERM_BYTES + 4];
    while (textBytes < limit && lr.next()) {
        char *s = lr.line.a; size_t n = lr.line.n;
        if (n && s[n - 1] == '\r') n--;
        // поля: источник \t id \t заголовок \t URL \t текст
        char *f[5]; size_t fl[5]; int k = 0; size_t st = 0;
        for (size_t i = 0; i <= n && k < 5; i++)
            if (i == n || (s[i] == '\t' && k < 4)) { f[k] = s + st; fl[k] = i - st; k++; st = i + 1; }
        if (k < 5) continue;
        docs++;
        double ts = now_sec();
        for (int part = 2; part <= 4; part += 2) {
            size_t pos = 0; int chars, len;
            textBytes += fl[part];
            while ((len = next_token(f[part], fl[part], pos, tok, chars)) > 0) {
                tokens++; tokenChars += chars; tokenBytes += len;
            }
        }
        tokenizeSec += now_sec() - ts;
    }
    fclose(in);
    double total = now_sec() - t0;
    double kb = textBytes / 1024.0;
    printf("documents          %llu\n", (unsigned long long)docs);
    printf("text_bytes         %llu\n", (unsigned long long)textBytes);
    printf("tokens             %llu\n", (unsigned long long)tokens);
    printf("avg_token_len      %.3f\n", tokens ? (double)tokenChars / tokens : 0.0);   // в символах
    printf("avg_token_bytes    %.3f\n", tokens ? (double)tokenBytes / tokens : 0.0);   // в байтах UTF-8
    printf("time_total_sec     %.3f\n", total);          // чтение файла + токенизация
    printf("time_tokenize_sec  %.3f\n", tokenizeSec);    // только токенизация
    printf("ms_per_kb          %.5f\n", kb ? tokenizeSec * 1000 / kb : 0.0);
    printf("mb_per_sec         %.2f\n", tokenizeSec ? textBytes / 1048576.0 / tokenizeSec : 0.0);
    return 0;
}

    std::vector<std::string> r;
    char tok[MAX_TERM_BYTES + 4];
    size_t pos = 0; int chars, len;
    while ((len = next_token(s.data(), s.size(), pos, tok, chars)) > 0) r.emplace_back(tok, len);
    return r;
}

static void show_tokens(const std::string &s) {
    auto t = tokens(s);
    printf("  текст:   %s\n  токены:  ", s.c_str());
    for (size_t i = 0; i < t.size(); i++) printf("%s%s", i ? " | " : "", t[i].c_str());
    printf("   (%zu шт.)\n", t.size());
}

static void cmd_text(const std::string &arg) {
    if (!arg.empty()) { show_tokens(arg); return; }
    printf("Вводите текст (пустая строка — выход):\n");
    std::string s;
    while (true) { printf("текст> "); fflush(stdout); if (!tool::read_line(s) || s.empty()) break; show_tokens(s); }
}

// Статистика: читаем corpus.tsv так же, как tokenize.exe --stats (лимит в байтах текста, 0 — весь корпус).
struct Stats { uint64_t docs = 0, bytes = 0, tokens = 0, chars = 0, tbytes = 0; double sec = 0; };
static Stats corpus_stats(double limitMb) {
    Stats st;
    uint64_t limit = limitMb > 0 ? (uint64_t)(limitMb * 1048576) : UINT64_MAX;
    char tok[MAX_TERM_BYTES + 4];
    tool::each_line(tool::path("data/corpus.tsv"), [&](std::string &line) {
        size_t f[6], k = 0; f[0] = 0;                        // начала полей
        for (size_t i = 0; i < line.size() && k < 4; i++) if (line[i] == '\t') f[++k] = i + 1;
        if (k < 4) return true;
        size_t t3 = line.find('\t', f[2]);                   // конец заголовка
        st.docs++;
        double t0 = now_sec();
        const char *parts[2] = {line.data() + f[2], line.data() + f[4]};
        size_t lens[2] = {t3 - f[2], line.size() - f[4]};
        for (int p = 0; p < 2; p++) {
            size_t pos = 0; int chars, len;
            st.bytes += lens[p];
            while ((len = next_token(parts[p], lens[p], pos, tok, chars)) > 0) { st.tokens++; st.chars += chars; st.tbytes += len; }
        }
        st.sec += now_sec() - t0;
        return st.bytes < limit;
    });
    return st;
}

static void cmd_stats(const std::string &) {
    if (!tool::exists(tool::path("data/corpus.tsv"))) { printf("нет data/corpus.tsv\n"); return; }
    printf("Токенизация всего корпуса...\n");
    Stats s = corpus_stats(0);
    printf("  документов                    %s\n", tool::num(s.docs).c_str());
    printf("  текст, байт                   %s\n", tool::num(s.bytes).c_str());
    printf("  токенов                       %s\n", tool::num(s.tokens).c_str());
    printf("  средняя длина токена, символов %.3f\n", (double)s.chars / s.tokens);
    printf("  средняя длина токена, байт     %.3f\n", (double)s.tbytes / s.tokens);
    printf("  время токенизации, с          %.3f\n", s.sec);
    printf("  мс на КБ                      %.5f\n", s.sec * 1000 / (s.bytes / 1024.0));
    printf("  МБ/с                          %.2f\n", s.bytes / 1048576.0 / s.sec);
}

static void cmd_speed(const std::string &) {
    printf("  %s %s %s %s\n", tool::pad("объём", 12).c_str(), tool::lpad("токенов", 14).c_str(), tool::lpad("время, с", 10).c_str(), tool::lpad("мс/КБ", 10).c_str());
    for (double mb : {5.0, 20.0, 80.0, 0.0}) {
        Stats s = corpus_stats(mb);
        printf("  %s %s %10.3f %10.5f\n", tool::pad(mb ? std::to_string((int)mb) + " МБ" : "весь корпус", 12).c_str(),
               tool::lpad(tool::num(s.tokens), 14).c_str(), s.sec, s.sec * 1000 / (s.bytes / 1024.0));
    }
    printf("\nВремя растёт линейно с объёмом (каждый байт обрабатывается один раз, O(n)); мс/КБ почти постоянно.\n");
}

static void cmd_utf8(const std::string &arg) {
    std::string s = arg.empty() ? "Жa€ё́" : arg;
    const unsigned char *u = (const unsigned char *)s.data();
    size_t i = 0;
    static const char *cls[] = {"разделитель", "часть слова", "пропускается (ударение)"};
    while (i < s.size()) {
        size_t st = i;
        uint32_t c = utf8_decode(u, s.size(), i);
        printf("  U+%04X  байты:", c);
        for (size_t k = st; k < i; k++) printf(" %02X", u[k]);
        printf("%*s  символ «%s» -> %s, нижний регистр U+%04X\n", (int)(4 - (i - st)) * 3, "",
               s.substr(st, i - st).c_str(), cls[char_class(c)], to_lower(c));
    }
}
