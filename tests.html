GoogleTest + CTest#

GoogleTest — фреймворк юнит-тестов от Google: тест описывается макросом TEST(Группа, Имя), проверки — EXPECT_EQ, EXPECT_TRUE, ASSERT_TRUE (ASSERT прерывает тест при неудаче, EXPECT — нет), GTEST_SKIP() пропускает тест (например, если нет полного корпуса).

CTest — запускатель тестов из CMake. В CMakeLists.txt строка gtest_discover_tests(${t} …) (в цикле по файлам тестов) регистрирует каждый TEST отдельным тестом CTest — в выводе видно каждый тест-кейс:

Как тестируются функции, а не только вывод#

Исходники программ подключаются в тест напрямую, а их main переименовываются макросом:

#define main indexer_main
#include "../src/indexer.cpp"
#undef main

Так тест вызывает next_token, stem::stem, TermTable::get_or_add, merge_sort_terms, op_and, Parser… и даже весь индексатор (indexer_main(argc, argv)) для мини-корпуса.

Проверка на полном корпусе — почему ей можно верить#

Эталон устроен совсем иначе, чем индекс: для каждого документа хранится отсортированный массив его термов; ответ на запрос — перебор всех документов, для каждого дерево запроса вычисляется рекурсивно, наличие терма — двоичным поиском. В эталоне нет ни списков документов, ни операций пересечения/объединения, поэтому ошибка в индексе или поиске не может «совпасть» с ошибкой эталона. Проверено, что тест умеет падать: на подменённом индексе совпало только 17 из 50 запросов.

static int full_corpus(const char *corpus, const char *indexPath, bool useStem, int nq) {
    int g_checks = 0, g_failed_checks = 0;
    printf("Full-corpus check: %s vs %s%s, %d random queries\n", corpus, indexPath,
           useStem ? " (stemmed)" : "", nq);
    double t0 = now_sec();
    FILE *in = fopen(corpus, "rb");
    if (!in) { printf("cannot open %s\n", corpus); return 1; }
    FullCtx c;
    {
        LineReader lr(in);
        Vec<uint32_t> cur;
        char tok[MAX_TERM_BYTES + 4], sb[MAX_TERM_BYTES + 8];
        bool isNew;
        while (lr.next()) {
            char *s = lr.line.a; size_t n = lr.line.n;
            if (n && s[n - 1] == '\r') n--;
            char *f[5]; size_t fl[5]; int k = 0; size_t st = 0;
            for (size_t i = 0; i <= n && k < 5; i++)
                if (i == n || (s[i] == '\t' && k < 4)) { f[k] = s + st; fl[k] = i - st; k++; st = i + 1; }
            if (k < 5) continue;
            cur.clear();
            for (int part = 2; part <= 4; part += 2) {           // заголовок и текст — как в индексаторе
                size_t pos = 0; int chars, len;
                while ((len = next_token(f[part], fl[part], pos, tok, chars)) > 0) {
                    uint32_t w = c.words.get_or_add(tok, len, isNew);
                    if (isNew) {
                        uint32_t key;
                        if (useStem) { int sl = stem::stem(tok, len, sb); key = c.keys.get_or_add(sb, sl, isNew); }
                        else key = c.keys.get_or_add(tok, len, isNew);
                        if (isNew) { c.keyWord.push(w); c.df.push(0); }
                        c.wordKey.push(key);
                    }
                    cur.push(c.wordKey[w]);
                }
            }
            qsort(cur.a, cur.n, 4, cmp_u32);
            size_t u = 0;
            for (size_t i = 0; i < cur.n; i++) if (!u || cur[u - 1] != cur[i]) cur[u++] = cur[i];
            DocTerms d; d.n = (uint32_t)u; d.t = (uint32_t *)malloc(u * 4 + 4);
            if (u) memcpy(d.t, cur.a, u * 4);
            for (size_t i = 0; i < u; i++) c.df[d.t[i]]++;
            c.docs.push(d);
        }
        cur.release();
    }
    fclose(in);
    printf("reference: %zu docs, %zu distinct keys, built in %.1f s\n", c.docs.n, c.keys.size, now_sec() - t0);

    Vec<uint32_t> order; order.resize(c.keys.size);
    for (uint32_t i = 0; i < c.keys.size; i++) order[i] = i;
    g_df = c.df.a;
    qsort(order.a, order.n, 4, cmp_df_desc);
    for (size_t i = 0; i < 300 && i < order.n; i++) c.frequent.push(order[i]);

    Index idx;
    if (!idx.load(indexPath)) return 1;
    EXPECT_EQ(idx.nDocs, c.docs.n);
    EXPECT_EQ(idx.stemmed, useStem);
    EXPECT_EQ(idx.nTerms, c.keys.size);
    int okCount = 0, bad = g_failed_checks;
    Vec<char> q; Vec<uint32_t> got, exp;
    double tq = 0;
    for (int i = 0; i < nq; i++) {
        q.clear(); c.nodes.clear();
        int root = gen_query(c, 0, q);
        exp.clear();
        for (uint32_t d = 0; d < c.docs.n; d++) if (eval_doc(c, root, c.docs[d])) exp.push(d);
        q.push(0);
        double ts = now_sec();
        run_query(idx, q.a, got);
        tq += now_sec() - ts;
        bool same = got.n == exp.n && (got.n == 0 || memcmp(got.a, exp.a, got.n * 4) == 0);
        g_checks++;
        if (same) okCount++;
        else {
            bad++; g_failed_checks++;
            if (bad <= 5) printf("    FAIL query \"%s\": index %zu docs, reference %zu\n", q.a, got.n, exp.n);
        }
    }
    printf("%d/%d random queries match the reference over all %zu documents (search time %.3f s total)\n",
           okCount, nq, c.docs.n, tq);
    printf(bad ? "FULL-CORPUS CHECK FAILED\n" : "FULL-CORPUS CHECK PASSED\n");
    return bad ? 1 : 0;
}

// Все токены строки, через пробел (для удобного сравнения).
static int tokens_joined(const char *s, char *out) {
    size_t pos = 0; int chars, len, n = 0;
    char tok[MAX_TERM_BYTES + 4];
    while ((len = next_token(s, strlen(s), pos, tok, chars)) > 0) {
        if (n) out[n++] = ' ';
        memcpy(out + n, tok, len); n += len;
    }
    out[n] = 0;
    return n;
}

static int stem_of(const char *w, char *out) { return stem::stem(w, (int)strlen(w), out); }

// Полный цикл поиска как в search.exe: лексер -> парсер -> вычисление. Результат — номера документов.
static void run_query(const Index &idx, const char *q, Vec<uint32_t> &res) {
    Vec<QTok> toks;
    lex(idx, q, strlen(q), toks);
    Parser p(toks);
    int root = p.parse();
    res.clear();
    if (root >= 0) eval(idx, p.nodes, root, res);
}

// Сравнивает результат запроса с ожидаемым списком документов (отсортированный, через запятую).
static bool same_docs(const Vec<uint32_t> &r, const uint32_t *exp, size_t n) {
    if (r.n != n) return false;
    for (size_t i = 0; i < n; i++) if (r[i] != exp[i]) return false;
    return true;
}
// Ожидаемый список документов сравнивается с выдачей поиска; при ошибке gtest печатает оба списка.
static std::string docs_str(const uint32_t *a, size_t n) {
    std::string s = "[";
    for (size_t i = 0; i < n; i++) { if (i) s += ","; s += std::to_string(a[i]); }
    return s + "]";
}

    char a0[] = "indexer", a3[] = "--stem";

// Проверка индекса и поиска на ПОЛНОМ корпусе (65 509 документов) против перебора.
#include "test_common.h"

// ================================================================ проверка на ПОЛНОМ корпусе
// unit_tests --full data\corpus.tsv data\index.bin [--stem] [N]
// Эталон строится независимо от индекса: для каждого документа — отсортированный массив его термов
// (qsort из стандартной библиотеки C). Ожидаемый ответ вычисляется перебором ВСЕХ документов:
// дерево запроса вычисляется для каждого документа рекурсивно, наличие терма — двоичным поиском.
// Списков документов и операций И/ИЛИ над ними в эталоне нет, поэтому ошибка индекса или поиска
// не может «совпасть» с ошибкой эталона.
struct DocTerms { uint32_t *t; uint32_t n; };
static int cmp_u32(const void *a, const void *b) {
    uint32_t x = *(const uint32_t *)a, y = *(const uint32_t *)b; return x < y ? -1 : x > y;
}
static const uint32_t *g_df = nullptr;
static int cmp_df_desc(const void *a, const void *b) {
    uint32_t i = *(const uint32_t *)a, j = *(const uint32_t *)b;
    if (g_df[i] != g_df[j]) return g_df[i] > g_df[j] ? -1 : 1;
    return i < j ? -1 : i > j;
}
static bool doc_has(const DocTerms &d, uint32_t term) {
    long lo = 0, hi = (long)d.n - 1;
    while (lo <= hi) {
        long m = (lo + hi) / 2;
        if (d.t[m] == term) return true;
        if (d.t[m] < term) lo = m + 1; else hi = m - 1;
    }
    return false;
}

enum QType { Q_TERM, Q_AND, Q_OR, Q_NOT };
struct QNode { QType type; int l, r; uint32_t term; };
struct Rng {
    uint64_t s;
    uint32_t next() { s ^= s << 13; s ^= s >> 7; s ^= s << 17; return (uint32_t)s; }
    double uni() { return (next() & 0xFFFFFF) / 16777216.0; }
};

struct FullCtx {
    TermTable words;              // словоформы корпуса
    TermTable keys;               // ключи эталона: словоформы или их основы (--stem)
    Vec<uint32_t> wordKey;        // словоформа -> ключ
    Vec<uint32_t> keyWord;        // ключ -> одна из его словоформ (её и пишем в запрос)
    Vec<uint32_t> df;             // в скольких документах встречается ключ
    Vec<DocTerms> docs;
    Vec<uint32_t> frequent;       // 300 самых частых ключей — чтобы пересечения были непустыми
    Vec<QNode> nodes;
    Rng rng{2026};
};

static int gen_query(FullCtx &c, int depth, Vec<char> &q) {
    double r = c.rng.uni();
    if (depth > 2 || r < 0.35) {
        uint32_t k = c.rng.uni() < 0.7 ? c.frequent[c.rng.next() % c.frequent.n]
                                       : (uint32_t)(c.rng.next() % c.keys.size);
        uint32_t w = c.keyWord[k];
        q.append(c.words.pool.a + c.words.off[w], c.words.len[w]);
        c.nodes.push({Q_TERM, -1, -1, k}); return (int)c.nodes.n - 1;
    }
    if (r < 0.45) {
        q.append("!(", 2); int x = gen_query(c, depth + 1, q); q.push(')');
        c.nodes.push({Q_NOT, x, -1, 0}); return (int)c.nodes.n - 1;
    }
    static const char *ops[] = {" ", " && ", "&&", "   ", " || ", "||"};
    bool isAnd = r < 0.75;
    q.push('(');
    int a = gen_query(c, depth + 1, q);
    const char *op = isAnd ? ops[c.rng.next() % 4] : ops[4 + c.rng.next() % 2];
    q.append(op, strlen(op));
    int b = gen_query(c, depth + 1, q);
    q.push(')');
    c.nodes.push({isAnd ? Q_AND : Q_OR, a, b, 0}); return (int)c.nodes.n - 1;
}

static bool eval_doc(const FullCtx &c, int v, const DocTerms &d) {
    const QNode &n = c.nodes[v];
    switch (n.type) {
        case Q_TERM: return doc_has(d, n.term);
        case Q_NOT:  return !eval_doc(c, n.l, d);
        case Q_AND:  return eval_doc(c, n.l, d) && eval_doc(c, n.r, d);
        default:     return eval_doc(c, n.l, d) || eval_doc(c, n.r, d);
    }
}

static int full_corpus(const char *corpus, const char *indexPath, bool useStem, int nq) {
    int g_checks = 0, g_failed_checks = 0;
    printf("Full-corpus check: %s vs %s%s, %d random queries\n", corpus, indexPath,
           useStem ? " (stemmed)" : "", nq);
    double t0 = now_sec();
    FILE *in = fopen(corpus, "rb");
    if (!in) { printf("cannot open %s\n", corpus); return 1; }
    FullCtx c;
    {
        LineReader lr(in);
        Vec<uint32_t> cur;
        char tok[MAX_TERM_BYTES + 4], sb[MAX_TERM_BYTES + 8];
        bool isNew;
        while (lr.next()) {
            char *s = lr.line.a; size_t n = lr.line.n;
            if (n && s[n - 1] == '\r') n--;
            char *f[5]; size_t fl[5]; int k = 0; size_t st = 0;
            for (size_t i = 0; i <= n && k < 5; i++)
                if (i == n || (s[i] == '\t' && k < 4)) { f[k] = s + st; fl[k] = i - st; k++; st = i + 1; }
            if (k < 5) continue;
            cur.clear();
            for (int part = 2; part <= 4; part += 2) {           // заголовок и текст — как в индексаторе
                size_t pos = 0; int chars, len;
                while ((len = next_token(f[part], fl[part], pos, tok, chars)) > 0) {
                    uint32_t w = c.words.get_or_add(tok, len, isNew);
                    if (isNew) {
                        uint32_t key;
                        if (useStem) { int sl = stem::stem(tok, len, sb); key = c.keys.get_or_add(sb, sl, isNew); }
                        else key = c.keys.get_or_add(tok, len, isNew);
                        if (isNew) { c.keyWord.push(w); c.df.push(0); }
                        c.wordKey.push(key);
                    }
                    cur.push(c.wordKey[w]);
                }
            }
            qsort(cur.a, cur.n, 4, cmp_u32);
            size_t u = 0;
            for (size_t i = 0; i < cur.n; i++) if (!u || cur[u - 1] != cur[i]) cur[u++] = cur[i];
            DocTerms d; d.n = (uint32_t)u; d.t = (uint32_t *)malloc(u * 4 + 4);
            if (u) memcpy(d.t, cur.a, u * 4);
            for (size_t i = 0; i < u; i++) c.df[d.t[i]]++;
            c.docs.push(d);
        }
        cur.release();
    }
    fclose(in);
    printf("reference: %zu docs, %zu distinct keys, built in %.1f s\n", c.docs.n, c.keys.size, now_sec() - t0);

    Vec<uint32_t> order; order.resize(c.keys.size);
    for (uint32_t i = 0; i < c.keys.size; i++) order[i] = i;
    g_df = c.df.a;
    qsort(order.a, order.n, 4, cmp_df_desc);
    for (size_t i = 0; i < 300 && i < order.n; i++) c.frequent.push(order[i]);

    Index idx;
    if (!idx.load(indexPath)) return 1;
    EXPECT_EQ(idx.nDocs, c.docs.n);
    EXPECT_EQ(idx.stemmed, useStem);
    EXPECT_EQ(idx.nTerms, c.keys.size);
    int okCount = 0, bad = g_failed_checks;
    Vec<char> q; Vec<uint32_t> got, exp;
    double tq = 0;
    for (int i = 0; i < nq; i++) {
        q.clear(); c.nodes.clear();
        int root = gen_query(c, 0, q);
        exp.clear();
        for (uint32_t d = 0; d < c.docs.n; d++) if (eval_doc(c, root, c.docs[d])) exp.push(d);
        q.push(0);
        double ts = now_sec();
        run_query(idx, q.a, got);
        tq += now_sec() - ts;
        bool same = got.n == exp.n && (got.n == 0 || memcmp(got.a, exp.a, got.n * 4) == 0);
        g_checks++;
        if (same) okCount++;
        else {
            bad++; g_failed_checks++;
            if (bad <= 5) printf("    FAIL query \"%s\": index %zu docs, reference %zu\n", q.a, got.n, exp.n);
        }
    }
    printf("%d/%d random queries match the reference over all %zu documents (search time %.3f s total)\n",
           okCount, nq, c.docs.n, tq);
    printf(bad ? "FULL-CORPUS CHECK FAILED\n" : "FULL-CORPUS CHECK PASSED\n");
    return bad ? 1 : 0;
}

TEST(Lab1_Corpus, all_documents_valid_and_unique) {
    tool::init_console();                                     // рабочая папка = lab1
    if (!tool::exists("data/ruwiki_text.jsonl")) GTEST_SKIP() << "нет корпуса в lab1/data";
    std::set<std::string> ids;
    size_t n = 0, empty = 0, noText = 0, dup = 0;
    each_doc([&](const Doc &d) {
        n++;
        if (d.title.empty() || d.url.rfind("https://", 0) != 0) empty++;
        if (d.text.empty()) noText++;     // статьи только из шаблонов (таблицы станций): после очистки пусто
        if (!ids.insert(d.source + ":" + d.id).second) dup++;
        return true;
    });
    EXPECT_EQ(n, 65509u) << "число документов корпуса";
    EXPECT_EQ(empty, 0u) << "документы без заголовка / URL";
    EXPECT_LE(noText, n / 1000) << "пустых после очистки документов не больше 0,1%";
    EXPECT_EQ(dup, 0u) << "повторяющиеся id";
}
