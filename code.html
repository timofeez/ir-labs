import json, os, sys, time
import requests

UA = {'User-Agent': 'MAI-IR-lab/1.0 (https://github.com/toshunster/MAI-IR; educational)'}

SOURCES = {
    'ruwiki': ('https://ru.wikipedia.org/w/api.php', 'Категория:Железнодорожный транспорт', 4),
    'enwiki': ('https://en.wikipedia.org/w/api.php', 'Category:Rail transport', 3),
}
MAX_CATS = 4000
OUT = os.path.join(os.path.dirname(__file__), '..', 'data')


def api(url, **p):
    p.update(format='json', formatversion=2)
    for attempt in range(8):
        try:
            r = requests.get(url, params=p, headers=UA, timeout=60)
            if r.status_code == 200:
                return r.json()
        except requests.RequestException:
            pass
        time.sleep(min(60, 3 * 2 ** attempt))
    raise RuntimeError('API failed')


def members(url, cat):
    cont = {}
    while True:
        d = api(url, action='query', list='categorymembers', cmtitle=cat,
                cmlimit=500, cmtype='page|subcat', **cont)
        yield from d['query']['categorymembers']
        if 'continue' not in d:
            return
        cont = d['continue']


def collect_titles(name, max_cats):
    url, root, depth = SOURCES[name]
    seen_cat, pages = {root}, {}
    level = [root]
    for dep in range(depth + 1):
        nxt = []
        for cat in level:
            for m in members(url, cat):
                if m['ns'] == 0:
                    pages.setdefault(m['pageid'], m['title'])
                elif m['ns'] == 14 and m['title'] not in seen_cat and len(seen_cat) < max_cats:
                    seen_cat.add(m['title']); nxt.append(m['title'])
        print(f'{name}: depth {dep}, cats {len(seen_cat)}, pages {len(pages)}', flush=True)
        level = nxt
    return pages, len(seen_cat)


if __name__ == '__main__':
    max_cats = int(sys.argv[1]) if len(sys.argv) > 1 else MAX_CATS
    out = sys.argv[2] if len(sys.argv) > 2 else OUT
    os.makedirs(out, exist_ok=True)
    stats = {}
    for name in SOURCES:
        pages, ncat = collect_titles(name, max_cats)
        with open(os.path.join(out, f'{name}_titles.tsv'), 'w', encoding='utf-8') as f:
            for pid, t in pages.items():
                f.write(f'{pid}\t{t}\n')
        stats[name] = {'categories': ncat, 'articles': len(pages)}
    with open(os.path.join(out, 'crawl_stats.json'), 'w', encoding='utf-8') as f:
        json.dump(stats, f, ensure_ascii=False, indent=2)

def get(url):
    for attempt in range(6):
        try:
            r = session.get(url, timeout=60)
            if r.status_code == 200:
                return r
            if r.status_code == 404:
                return None
        except requests.RequestException:
            pass
        time.sleep(min(60, 3 * 2 ** attempt))
    return None


def list_articles(limit):
    index = get(SITEMAP_INDEX).text
    parts = re.findall(r'<loc>([^<]+\.xml)</loc>', index)
    items = []
    for p in parts:
        items += ARTICLE_RE.findall(get(p).text)
        time.sleep(DELAY)
    items.sort(key=lambda t: (t[2], int(t[1])), reverse=True)
    return items[:limit]

REF = re.compile(r'<ref[^>/]*/>|<ref[^>]*>.*?</ref>', re.S | re.I)
COMMENT = re.compile(r'<!--.*?-->', re.S)
TAG = re.compile(r'<[^>]+>')
TABLE = re.compile(r'\{\|.*?\|\}', re.S)
FILE = re.compile(r'\[\[(?:Файл|Изображение|File|Image|Категория|Category):[^\[\]]*(?:\[\[[^\]]*\]\][^\[\]]*)*\]\]', re.I)
LINK = re.compile(r'\[\[(?:[^|\]]*\|)?([^\]]*)\]\]')
EXTLINK = re.compile(r'\[https?://\S+\s*([^\]]*)\]')
TEMPLATE = re.compile(r'\{\{[^{}]*\}\}')
HEADING = re.compile(r'^=+\s*(.*?)\s*=+\s*$', re.M)
FORMAT = re.compile(r"'{2,}")
TAIL = re.compile(r'^==\s*(Примечания|Литература|Ссылки|См\. также|References|External links|See also|Further reading|Notes|Bibliography)\s*==.*',
                  re.S | re.M)
WORD = re.compile(r'\w+', re.U)
RUBRIC = re.compile(r'^\s*([^|\n]{2,40}?)\s*\|\s*')


def clean(wt):
    t = TAIL.sub('', wt)
    t = COMMENT.sub('', t)
    t = REF.sub('', t)
    for _ in range(5):
        t2 = TEMPLATE.sub('', t)
        if t2 == t:
            break
        t = t2
    t = TABLE.sub('', t)
    t = FILE.sub('', t)
    t = LINK.sub(r'\1', t)
    t = EXTLINK.sub(r'\1', t)
    t = TAG.sub('', t)
    t = HEADING.sub(r'\1', t)
    t = FORMAT.sub('', t)
    t = re.sub(r'^[*#:;]+\s*', '', t, flags=re.M)
    t = re.sub(r'\n{3,}', '\n\n', t)
    return t.strip()


def clean_html(html):
    soup = BeautifulSoup(html, 'lxml')
    body = soup.select_one('article.article-block__main-body')
    header = soup.select_one('div.article-header')
    meta = {
        'issue': bool(header and header.find('h2') and 'Выпуск' in header.find('h2').get_text()),
        'photo': bool(header and header.select_one('.article-header__img img')),
        'subheadings': bool(body and body.find('b')),
        'author': bool(body and body.select_one('div.avt')),
        'rubric': False,
    }
    if body is None:
        return '', meta
    for el in body.select('div.avt, script, style'):
        el.decompose()
    for br in body.find_all('br'):
        br.replace_with('\n')
    text = body.get_text()
    m = RUBRIC.match(text)
    if m:
        meta['rubric'] = True
        text = text[m.end():]
    text = re.sub(r'[ \t\xa0]+', ' ', text)
    text = re.sub(r'\n\s*\n\s*(\n\s*)+', '\n\n', text)
    paragraphs = text.strip().split('\n\n')
    bold = body.find_all('b')
    last = paragraphs[-1].strip()
    if bold and bold[-1].get_text(strip=True) and bold[-1].get_text(strip=True) in last \
            and len(last.split()) <= 20 and len(paragraphs) > 1:
        meta['author'] = True
        paragraphs = paragraphs[:-1]
    return '\n\n'.join(p.strip() for p in paragraphs).strip(), meta

static std::string json_str(const std::string &j, const std::string &key) {
    std::string pat = "\"" + key + "\": ";
    size_t p = j.find(pat);
    if (p == std::string::npos) return "";
    p += pat.size();
    if (j[p] != '"') {
        size_t e = j.find_first_of(",}", p);
        return j.substr(p, e - p);
    }
    std::string r;
    for (size_t i = p + 1; i < j.size(); i++) {
        char c = j[i];
        if (c == '"') break;
        if (c != '\\') { r += c; continue; }
        c = j[++i];
        if (c == 'n') r += '\n'; else if (c == 't') r += '\t'; else if (c == 'r') r += '\r';
        else if (c == 'u') {
            uint32_t cp = (uint32_t)strtoul(j.substr(i + 1, 4).c_str(), nullptr, 16); i += 4;
            if (cp >= 0xD800 && cp < 0xDC00 && j.compare(i + 1, 2, "\\u") == 0) {
                uint32_t lo = (uint32_t)strtoul(j.substr(i + 3, 4).c_str(), nullptr, 16); i += 6;
                cp = 0x10000 + ((cp - 0xD800) << 10) + (lo - 0xDC00);
            }
            char b[4]; int n = utf8_encode(cp, b); r.append(b, n);
        } else r += c;
    }
    return r;
}

    static const regex HEADING("(^|\n)=+\\s*([^\n]*?)\\s*=+[ \t]*(?=\n|$)");
    static const regex FORMAT("'{2,}");
    static const regex LIST("(^|\n)[*#:;]+[ \t]*");
    static const regex BLANKS("\n{3,}");
    t = regex_replace(t, COMMENT, "");
    t = regex_replace(t, REF, "");
    for (int k = 0; k < 5; k++) { std::string t2 = regex_replace(t, TEMPLATE, ""); if (t2 == t) break; t = t2; }
    t = regex_replace(t, TABLE, "");
    t = regex_replace(t, FILE_, "");
    t = regex_replace(t, LINK, "$1");
    t = regex_replace(t, EXTLINK, "$1");
    t = regex_replace(t, TAG, "");
    t = regex_replace(t, HEADING, "$1$2");
    t = regex_replace(t, FORMAT, "");
    t = regex_replace(t, LIST, "$1");
    t = regex_replace(t, BLANKS, "\n\n");
    size_t a = t.find_first_not_of(" \n\t"), b = t.find_last_not_of(" \n\t");
    return a == std::string::npos ? "" : t.substr(a, b - a + 1);
}

        self.max_requests = int(logic.get('max_requests', 0))
        self.respect_robots = bool(logic.get('respect_robots', True))
        self.robots = {}
        self.sources = {s['name']: s for s in cfg['sources']}
        self.db = Db(cfg['db'])
        print(f'БД: CockroachDB, база {self.db.database}, подключение к узлу {self.db.node}', flush=True)
        self.http = requests.Session()
        self.http.headers['User-Agent'] = logic.get('user_agent', 'MAI-IR-robot/1.0')
        self.requests_done = 0
        self.stop = False
        self.last_request = 0.0

    def selector(self, source):
        return self.sources.get(source, {}).get('content_selector')

    def get(self, url, headers=None):
        for attempt in range(self.retries):
            wait = self.delay - (time.perf_counter() - self.last_request)
            if wait > 0:
                time.sleep(wait)
            self.last_request = time.perf_counter()
            self.requests_done += 1
            try:
                r = self.http.get(url, headers=headers or {}, timeout=self.timeout)
                if r.status_code in (200, 304, 404):
                    return r
                if r.status_code == 429 or r.status_code >= 500:
                    time.sleep(min(60, 3 * 2 ** attempt))
                    continue
                return r
            except requests.RequestException:
                time.sleep(min(60, 3 * 2 ** attempt))
        return None

    def allowed(self, url):
        if not self.respect_robots:
            return True
        p = urlsplit(url)
        key = f'{p.scheme}://{p.netloc}'
        if key not in self.robots:
            rp = urllib.robotparser.RobotFileParser()
            r = self.get(key + '/robots.txt')
            if r is not None and r.status_code == 200:
                rp.parse(r.text.splitlines())
            else:
                rp.allow_all = True
            cd = rp.crawl_delay(self.http.headers['User-Agent'])
            if cd:
                self.delay = max(self.delay, float(cd))
            self.robots[key] = rp
        return self.robots[key].can_fetch(self.http.headers['User-Agent'], url)

    def read_sitemap(self, src):
        pattern = re.compile(src['url_pattern'])
        todo, seen, found = [src['sitemap']], set(), {}
        while todo and not self.stop:
            sm = todo.pop(0)
            if sm in seen or not self.allowed(sm):
                continue
            seen.add(sm)
            r = self.get(sm)
            if r is None or r.status_code != 200:
                continue
            children, urls = parse_sitemap(r.text)
            todo += children
            for u, lastmod in urls:
                if pattern.search(u):
                    found[normalize_url(u)] = lastmod
        return found

    def enqueue(self, url, source, kind, depth, is_cont=0, lastmod=None):
        self.db.execute('INSERT INTO frontier(url, source, kind, depth, is_cont, lastmod) '
                        'VALUES (?,?,?,?,?,?) ON CONFLICT (url) DO NOTHING',
                        (url, source, kind, depth, is_cont, lastmod))

    def categories_count(self, source):
        return self.db.execute("SELECT COUNT(*) FROM frontier WHERE source=? AND kind='category' AND is_cont=0",
                               (source,)).fetchone()[0]

    def seed(self):
        for s in self.sources.values():
            if s.get('type', 'categories') == 'sitemap':
                if self.db.execute('SELECT 1 FROM frontier WHERE source=? LIMIT 1', (s['name'],)).fetchone():
                    continue
                found = self.read_sitemap(s)
                items = sorted(found.items(), key=lambda t: (t[1], t[0]), reverse=True)
                for url, lastmod in items[:int(s.get('max_documents', len(items)))]:
                    self.enqueue(url, s['name'], 'article', 0, lastmod=lastmod)
                print(f'{s["name"]}: из sitemap взято {min(len(items), int(s.get("max_documents", len(items))))} '
                      f'из {len(items)} подходящих адресов', flush=True)
            else:
                for u in s['seeds']:
                    self.enqueue(normalize_url(u), s['name'], 'category', 0)
        self.db.commit()

    def process_category(self, row, html):
        url, source, depth = row[1], row[2], row[4]
        src = self.sources[source]
        soup = BeautifulSoup(html, 'lxml')
        host = urlsplit(url).hostname
        for block_id, kind in (('mw-subcategories', 'category'), ('mw-pages', 'article')):
            block = soup.find(id=block_id)
            if not block:
                continue
            for a in block.find_all('a', href=True):
                link = normalize_url(a['href'], url)
                if urlsplit(link).hostname != host:
                    continue
                text = a.get_text(strip=True).lower()
                if text in ('следующая страница', 'next page'):
                    self.enqueue(link, source, 'category', depth, is_cont=1)
                    continue
                if text in ('предыдущая страница', 'previous page') or not urlsplit(link).path.startswith('/wiki/'):
                    continue
                if a.find_parent(class_='redirect-in-category'):
                    continue
                if kind == 'category':
                    if depth + 1 <= src['max_depth'] and self.categories_count(source) < src['max_categories']:
                        self.enqueue(link, source, 'category', depth + 1)
                elif not title_of(link).startswith(SKIP_PREFIXES):
                    self.enqueue(link, source, 'article', depth + 1)

    @staticmethod
    def canonical(url, r):
        m = re.search(rb'<link rel="canonical" href="([^"]+)"', r.content[:20000])
        return normalize_url(m.group(1).decode('utf-8', 'replace'), url) if m else url

    def duplicate_of(self, url, source, r):
        canon = self.canonical(url, r)
        if canon != url and self.db.execute('SELECT 1 FROM documents WHERE url=?', (canon,)).fetchone():
            return canon
        row = self.db.execute('SELECT url FROM documents WHERE content_hash=? AND url<>?',
                              (content_hash(r.content, self.selector(source)), canon)).fetchone()
        return row[0] if row else None

    @staticmethod
    def page_title(html, url):
        soup = BeautifulSoup(html, 'lxml')
        h = soup.find(id='firstHeading')
        if h:
            return h.get_text(strip=True)
        og = soup.find('meta', property='og:title')
        if og and og.get('content'):
            return og['content'].strip()
        return title_of(url)

    def save_document(self, url, source, r, now, sitemap_lastmod=None):
        html = r.content
        h = content_hash(html, self.selector(source))
        blob = zlib.compress(html, 6) if self.compress else html
        self.db.execute(
            'INSERT INTO documents(url, source, title, html, compressed, html_size, content_hash, last_modified,'
            ' crawled_at, checked_at, sitemap_lastmod, hash_v) VALUES (?,?,?,?,?,?,?,?,?,?,?,?)'
            ' ON CONFLICT(url) DO UPDATE SET title=excluded.title, html=excluded.html, compressed=excluded.compressed,'
            ' html_size=excluded.html_size, content_hash=excluded.content_hash, last_modified=excluded.last_modified,'
            ' crawled_at=excluded.crawled_at, checked_at=excluded.checked_at, sitemap_lastmod=excluded.sitemap_lastmod,'
            ' hash_v=excluded.hash_v, versions=documents.versions+1',
            (url, source, self.page_title(html, url), blob, int(self.compress), len(html), h,
             r.headers.get('Last-Modified'), now, now, sitemap_lastmod, HASH_V))

    def crawl_frontier(self):
        did = False
        while not self.stop:
            row = self.db.execute("SELECT id, url, source, kind, depth, is_cont, lastmod FROM frontier "
                                  "WHERE state='new' ORDER BY id LIMIT 1").fetchone()
            if not row:
                self.db.commit()
                return did
            did = True
            fid, url, source, kind, depth, _, lastmod = row
            if kind == 'article' and self.db.execute('SELECT 1 FROM documents WHERE url=?', (url,)).fetchone():
                self.db.execute("UPDATE frontier SET state='done' WHERE id=?", (fid,))
                self.db.commit()
                continue
            if not self.allowed(url):
                self.db.execute("UPDATE frontier SET state='skipped', error='robots.txt' WHERE id=?", (fid,))
                self.db.commit()
                continue
            r = self.get(url)
            if r is None or r.status_code != 200:
                err = 'no response' if r is None else f'HTTP {r.status_code}'
                self.db.execute("UPDATE frontier SET state='error', error=? WHERE id=?", (err, fid))
            elif kind == 'category':
                self.process_category(row, r.text)
                self.db.execute("UPDATE frontier SET state='done' WHERE id=?", (fid,))
            else:
                dup = self.duplicate_of(url, source, r)
                if dup:
                    self.db.execute("UPDATE frontier SET state='duplicate', error=? WHERE id=?", (dup, fid))
                else:
                    self.save_document(self.canonical(url, r), source, r, int(time.time()), lastmod)
                    self.db.execute("UPDATE frontier SET state='done' WHERE id=?", (fid,))
            self.db.commit()
            self.progress(kind, url)
            if self.max_requests and self.requests_done >= self.max_requests:
                self.stop = True

    def stored_hash(self, url, source, old_hash, hash_v):
        if hash_v == HASH_V:
            return old_hash
        blob, compressed = self.db.execute('SELECT html, compressed FROM documents WHERE url=?', (url,)).fetchone()
        h = content_hash(zlib.decompress(blob) if compressed else blob, self.selector(source))
        self.db.execute('UPDATE documents SET content_hash=?, hash_v=? WHERE url=?', (h, HASH_V, url))
        return h

    def recrawl(self):
        did = False
        pass_start = int(time.time())
        due = pass_start - self.recrawl_interval
        fresh = {}
        for s in self.sources.values():
            if s.get('type') == 'sitemap' and self.db.execute(
                    'SELECT 1 FROM documents WHERE source=? AND checked_at <= ? AND checked_at < ? LIMIT 1',
                    (s['name'], due, pass_start)).fetchone():
                fresh[s['name']] = self.read_sitemap(s)
        while not self.stop:
            now = int(time.time())
            row = self.db.execute('SELECT url, source, last_modified, content_hash, hash_v, sitemap_lastmod '
                                  'FROM documents WHERE checked_at <= ? AND checked_at < ? ORDER BY checked_at LIMIT 1',
                                  (due, pass_start)).fetchone()
            if not row:
                self.db.commit()
                return did
            did = True
            url, source, last_modified, old_hash, hash_v, sm_old = row
            if source in fresh and fresh[source].get(url) == sm_old and sm_old:
                self.db.execute('UPDATE documents SET checked_at=? WHERE url=?', (now, url))
                status = 'sitemap: not modified'
            else:
                r = self.get(url, {'If-Modified-Since': last_modified} if last_modified else None)
                old = self.stored_hash(url, source, old_hash, hash_v)
                new_lastmod = fresh.get(source, {}).get(url, sm_old)
                if r is not None and r.status_code == 200 and content_hash(r.content, self.selector(source)) != old:
                    self.save_document(url, source, r, now, new_lastmod)
                    status = 'updated'
                else:
                    self.db.execute('UPDATE documents SET checked_at=?, sitemap_lastmod=? WHERE url=?',
                                    (now, new_lastmod, url))
                    status = 'not modified' if r is not None and r.status_code == 304 else 'same content'
            self.db.commit()
            self.progress('recrawl ' + status, url)
            if self.max_requests and self.requests_done >= self.max_requests:
                self.stop = True

    def progress(self, what, url):
        if self.requests_done % 100 == 0 or self.cfg['logic'].get('verbose'):
            docs = self.db.execute('SELECT COUNT(*) FROM documents').fetchone()[0]
            queue = self.db.execute("SELECT COUNT(*) FROM frontier WHERE state='new'").fetchone()[0]
            print(f'[{time.strftime("%H:%M:%S")}] requests={self.requests_done} documents={docs} '
                  f'queue={queue} last={what} {unquote(url)}', flush=True)

    def run(self):
        self.seed()
        while not self.stop:
            busy = self.crawl_frontier()
            busy = self.recrawl() or busy
            if self.stop:
                break
            if not busy:
                if self.exit_when_idle:
                    print('Нечего обкачивать — выход.')
                    break
                time.sleep(self.idle_sleep)
        self.db.commit()
        self.db.close()
        self.http.close()
        print(f'Остановлен. Запросов за запуск: {self.requests_done}')

def connect_any(hosts, database, user, autocommit=False):
    last = None
    for h in hosts:
        host, _, port = h.partition(':')

def copy(src, db, table, cols, key, batch):
    total = src.execute(f'SELECT COUNT(*) FROM {table}').fetchone()[0]
    row_ph = '(' + ', '.join(['%s'] * len(cols)) + ')'
    cur = src.execute(f'SELECT {", ".join(cols)} FROM {table} ORDER BY rowid')
    done, t0 = 0, time.time()
    while True:
        rows = cur.fetchmany(batch)
        if not rows:
            break
        for attempt in range(20):
            try:
                sql = (f'INSERT INTO {table} ({", ".join(cols)}) VALUES {", ".join([row_ph] * len(rows))} '
                       f'ON CONFLICT ({key}) DO NOTHING')
                db.conn.execute(sql, [v for r in rows for v in r])
                db.commit()
                break
            except psycopg.Error as e:
                try:
                    db.rollback()
                except psycopg.Error:
                    db.reconnect()
                print(f'\n  повтор порции ({type(e).__name__}): {str(e)[:80]}', flush=True)
                time.sleep(min(60, 5 * (attempt + 1)))
        else:
            raise RuntimeError('порция не записана после 20 попыток')
        done += len(rows)
        print(f'\r{table}: {done}/{total} ({time.time() - t0:.0f} с)', end='', flush=True)
    print()
    n = db.execute(f'SELECT COUNT(*) FROM {table}').fetchone()[0]
    db.commit()
    print(f'{table}: в SQLite {total}, в CockroachDB {n}')
    return total, n

static std::string normalize_url(std::string url, const std::string &base = "") {

    }

    auto ends = [](const std::string &a, const std::string &b) { return a.size() >= b.size() && a.compare(a.size() - b.size(), b.size(), b) == 0; };

    std::replace(path.begin(), path.end(), ' ', '_');
    std::vector<std::pair<std::string, std::string>> kv;

static bool robots_allowed(const std::string &robots, const std::string &path, const std::string &agent = "MAI-IR-robot") {
    struct Rule { bool allow; std::string prefix; };
    std::vector<Rule> mine, star, *cur = nullptr;
    bool lastWasAgent = false;
    size_t p = 0;
    std::vector<std::vector<Rule> *> group;
    while (p <= robots.size()) {
        size_t q = robots.find('\n', p); if (q == std::string::npos) q = robots.size();
        std::string l = robots.substr(p, q - p); p = q + 1;
        size_t h = l.find('#'); if (h != std::string::npos) l.resize(h);
        while (!l.empty() && isspace((unsigned char)l.back())) l.pop_back();
        size_t c = l.find(':');
        if (c != std::string::npos) {
            std::string k = lower_ascii(l.substr(0, c)), v = l.substr(c + 1);
            while (!v.empty() && v[0] == ' ') v.erase(0, 1);
            if (k == "user-agent") {
                if (!lastWasAgent) group.clear();
                if (v == "*") group.push_back(&star);
                else if (lower_ascii(agent).find(lower_ascii(v)) != std::string::npos) group.push_back(&mine);
                lastWasAgent = true;
            } else {
                lastWasAgent = false;
                if ((k == "allow" || k == "disallow") && !v.empty())
                    for (auto *g : group) g->push_back({k == "allow", v});
            }
        }
        if (q == robots.size()) break;
    }
    cur = mine.empty() ? &star : &mine;
    int best = -1; bool allow = true;
    for (auto &r : *cur) {
        std::string pre = r.prefix; bool endAnchor = !pre.empty() && pre.back() == '$';
        if (endAnchor) pre.pop_back();
        size_t star_ = pre.find('*');
        bool match = star_ == std::string::npos ? (endAnchor ? path == pre : path.rfind(pre, 0) == 0)
                                                : path.rfind(pre.substr(0, star_), 0) == 0 && path.find(pre.substr(star_ + 1)) != std::string::npos;
        if (match && ((int)r.prefix.size() > best || ((int)r.prefix.size() == best && r.allow))) { best = (int)r.prefix.size(); allow = r.allow; }
    }
    return allow;
}

template <class T>
struct Vec {
    T *a = nullptr;
    size_t n = 0, cap = 0;

    void grow(size_t c) {
        T *b = (T *)realloc(a, c * sizeof(T));
        if (!b) { fprintf(stderr, "out of memory\n"); exit(1); }
        a = b; cap = c;
    }
    void reserve(size_t c) { if (c > cap) grow(c); }
    void push(const T &x) { if (n == cap) grow(cap ? cap * 2 : 16); a[n++] = x; }
    void append(const T *src, size_t k) {
        if (n + k > cap) { size_t c = cap ? cap : 16; while (c < n + k) c *= 2; grow(c); }
        memcpy(a + n, src, k * sizeof(T)); n += k;
    }
    void resize(size_t k) { reserve(k); n = k; }
    void clear() { n = 0; }
    void release() { free(a); a = nullptr; n = cap = 0; }
    T &operator[](size_t i) { return a[i]; }
    const T &operator[](size_t i) const { return a[i]; }

static inline uint32_t utf8_decode(const unsigned char *s, size_t n, size_t &i) {
    unsigned c = s[i];
    if (c < 0x80) { i++; return c; }
    int len = c >= 0xF0 ? 4 : c >= 0xE0 ? 3 : c >= 0xC0 ? 2 : 1;
    if (len == 1 || i + len > n) { i++; return 0xFFFD; }
    uint32_t cp = c & (0xFF >> (len + 1));
    for (int k = 1; k < len; k++) cp = (cp << 6) | (s[i + k] & 0x3F);
    i += len;
    return cp;
}

static inline int utf8_encode(uint32_t c, char *o) {

    if (c < 0x800) { o[0] = (char)(0xC0 | (c >> 6)); o[1] = (char)(0x80 | (c & 0x3F)); return 2; }
    if (c < 0x10000) {

static inline int next_token(const char *s, size_t n, size_t &pos, char *out, int &chars) {
    const unsigned char *u = (const unsigned char *)s;
    int len = 0; chars = 0;
    while (pos < n) {
        size_t start = pos;
        uint32_t c = utf8_decode(u, n, pos);
        int cls = char_class(c);
        if (cls == 1) {

        }
    }
    if (len > 0 && pos < n) { size_t p = pos; utf8_decode(u, n, p); pos = p; }
    return len;
}

                if (end == 0) { eof = true; return line.n > 0; }
            }
            char *p = (char *)memchr(buf + beg, '\n', end - beg);
            size_t stop = p ? (size_t)(p - buf) : end;
            line.append(buf + beg, stop - beg);
            beg = p ? stop + 1 : end;
            if (p) return true;
        }
    }
};

struct TermTable {
    struct Slot { uint32_t hash, id; };
    Slot *slots = nullptr;
    size_t cap = 0, size = 0;
    Vec<char> pool;
    Vec<uint32_t> off;
    Vec<uint8_t> len;

    static uint32_t fnv1a(const char *s, int n) {
        uint32_t h = 2166136261u;
        for (int i = 0; i < n; i++) { h ^= (unsigned char)s[i]; h *= 16777619u; }
        return h;
    }
    void rehash(size_t newcap) {
        Slot *ns = (Slot *)malloc(newcap * sizeof(Slot));
        for (size_t i = 0; i < newcap; i++) ns[i].id = UINT32_MAX;
        for (size_t i = 0; i < cap; i++) {
            if (slots[i].id == UINT32_MAX) continue;
            size_t j = slots[i].hash & (newcap - 1);
            while (ns[j].id != UINT32_MAX) j = (j + 1) & (newcap - 1);
            ns[j] = slots[i];
        }
        free(slots); slots = ns; cap = newcap;
    }
    uint32_t get_or_add(const char *s, int n, bool &isNew) {
        if (size * 2 >= cap) rehash(cap ? cap * 2 : 1 << 16);
        uint32_t h = fnv1a(s, n);
        size_t j = h & (cap - 1);
        while (slots[j].id != UINT32_MAX) {
            uint32_t id = slots[j].id;
            if (slots[j].hash == h && len[id] == n && memcmp(pool.a + off[id], s, n) == 0) {
                isNew = false; return id;
            }
            j = (j + 1) & (cap - 1);
        }
        uint32_t id = (uint32_t)size++;
        slots[j].hash = h; slots[j].id = id;
        off.push((uint32_t)pool.n); len.push((uint8_t)n); pool.append(s, n);
        isNew = true;
        return id;
    }
};

            while (i < mid && j < hi) {

                    dst[k++] = src[j++];

                    dst[k++] = src[i++];
            }

            while (j < hi) dst[k++] = src[j++];
        }

    }
    if (src != a) memcpy(a, src, n * sizeof(uint32_t));
    free(tmp);

static void op_and(const Vec<uint32_t> &a, const Vec<uint32_t> &b, Vec<uint32_t> &r) {
    r.clear(); size_t i = 0, j = 0;
    while (i < a.n && j < b.n) {
        if (a[i] < b[j]) i++;
        else if (a[i] > b[j]) j++;
        else { r.push(a[i]); i++; j++; }
    }
}
static void op_or(const Vec<uint32_t> &a, const Vec<uint32_t> &b, Vec<uint32_t> &r) {
    r.clear(); r.reserve(a.n + b.n); size_t i = 0, j = 0;
    while (i < a.n && j < b.n) {
        if (a[i] < b[j]) r.push(a[i++]);
        else if (a[i] > b[j]) r.push(b[j++]);
        else { r.push(a[i]); i++; j++; }
    }
    while (i < a.n) r.push(a[i++]);
    while (j < b.n) r.push(b[j++]);
}

enum TokType { T_TERM, T_AND, T_OR, T_NOT, T_LP, T_RP };
struct QTok { TokType type; long entry; };

static void lex(const Index &idx, const char *q, size_t n, Vec<QTok> &out) {
    out.clear();
    size_t i = 0;
    char tok[MAX_TERM_BYTES + 4], sb[MAX_TERM_BYTES + 8];
    while (i < n) {
        char c = q[i];
        if (c == ' ' || c == '\t' || c == '\r' || c == '\n') { i++; continue; }
        if (c == '(') { out.push({T_LP, 0}); i++; continue; }
        if (c == ')') { out.push({T_RP, 0}); i++; continue; }
        if (c == '!') { out.push({T_NOT, 0}); i++; continue; }
        if (c == '&') { while (i < n && q[i] == '&') i++; out.push({T_AND, 0}); continue; }
        if (c == '|') { while (i < n && q[i] == '|') i++; out.push({T_OR, 0}); continue; }
        size_t st = i;
        while (i < n && !strchr(" \t\r\n()!&|", q[i])) i++;
        size_t pos = 0; int chars, len;
        bool first = true;
        while ((len = next_token(q + st, i - st, pos, tok, chars)) > 0) {
            if (!first) out.push({T_AND, 0});
            if (idx.stemmed) {
                int sl = stem::stem(tok, len, sb);
                out.push({T_TERM, idx.find_term(sb, sl)});
            } else {
                out.push({T_TERM, idx.find_term(tok, len)});
            }
            first = false;
        }
    }
}

enum NodeType { N_TERM, N_AND, N_OR, N_NOT };
struct Node { NodeType type; int l, r; long entry; };

struct Parser {
    const Vec<QTok> &t;
    size_t p = 0;
    Vec<Node> nodes;
    explicit Parser(const Vec<QTok> &toks) : t(toks) {}

    int make(NodeType ty, int l, int r, long e = 0) { nodes.push({ty, l, r, e}); return (int)nodes.n - 1; }
    bool at(TokType ty) const { return p < t.n && t[p].type == ty; }

    int unary() {
        if (at(T_NOT)) { p++; int c = unary(); return c < 0 ? -1 : make(N_NOT, c, -1); }
        if (at(T_LP)) { p++; int e = expr(); if (at(T_RP)) p++; return e; }
        if (at(T_TERM)) { long e = t[p].entry; p++; return make(N_TERM, -1, -1, e); }
        return -1;
    }
    int conj() {
        int left = unary();
        for (;;) {
            if (at(T_AND)) { p++; continue; }
            if (at(T_TERM) || at(T_NOT) || at(T_LP)) {
                int right = unary();
                if (right < 0) continue;
                left = left < 0 ? right : make(N_AND, left, right);
                continue;
            }
            return left;
        }
    }
    int expr() {
        int left = conj();
        while (at(T_OR)) {
            p++;
            int right = conj();
            if (right < 0) continue;
            left = left < 0 ? right : make(N_OR, left, right);
        }
        return left;
    }
    int parse() {
        int root = expr();
        while (p < t.n) {
            p++;
            int more = expr();
            if (more >= 0) root = root < 0 ? more : make(N_AND, root, more);
        }
        return root;
    }
};

static void eval(const Index &idx, const Vec<Node> &nodes, int v, Vec<uint32_t> &res) {
    const Node &nd = nodes[v];
    if (nd.type == N_TERM) { idx.postings_of(nd.entry, res); return; }
    Vec<uint32_t> a, b;
    if (nd.type == N_NOT) { eval(idx, nodes, nd.l, a); op_not(a, idx.nDocs, res); a.release(); return; }
    if (nd.type == N_AND && nodes[nd.r].type == N_NOT) {
        eval(idx, nodes, nd.l, a); eval(idx, nodes, nodes[nd.r].l, b); op_and_not(a, b, res);
    } else if (nd.type == N_AND && nodes[nd.l].type == N_NOT) {
        eval(idx, nodes, nd.r, a); eval(idx, nodes, nodes[nd.l].l, b); op_and_not(a, b, res);
    } else {
        eval(idx, nodes, nd.l, a); eval(idx, nodes, nd.r, b);
        if (nd.type == N_AND) op_and(a, b, res); else op_or(a, b, res);
    }
    a.release(); b.release();
}

    int a = gen_query(c, depth + 1, q);
    const char *op = isAnd ? ops[c.rng.next() % 4] : ops[4 + c.rng.next() % 2];
    q.append(op, strlen(op));
    int b = gen_query(c, depth + 1, q);
    q.push(')');
    c.nodes.push({isAnd ? Q_AND : Q_OR, a, b, 0}); return (int)c.nodes.n - 1;
}

static bool eval_doc(const FullCtx &c, int v, const DocTerms &d) {
    const QNode &n = c.nodes[v];
    switch (n.type) {
        case Q_TERM: return doc_has(d, n.term);
        case Q_NOT:  return !eval_doc(c, n.l, d);
        case Q_AND:  return eval_doc(c, n.l, d) && eval_doc(c, n.r, d);
        default:     return eval_doc(c, n.l, d) || eval_doc(c, n.r, d);
    }
}

static int full_corpus(const char *corpus, const char *indexPath, bool useStem, int nq) {
    int g_checks = 0, g_failed_checks = 0;

           useStem ? " (stemmed)" : "", nq);

        bool isNew;
        while (lr.next()) {
            char *s = lr.line.a; size_t n = lr.line.n;
            if (n && s[n - 1] == '\r') n--;
            char *f[5]; size_t fl[5]; int k = 0; size_t st = 0;
            for (size_t i = 0; i <= n && k < 5; i++)
                if (i == n || (s[i] == '\t' && k < 4)) { f[k] = s + st; fl[k] = i - st; k++; st = i + 1; }
            if (k < 5) continue;
            cur.clear();
            for (int part = 2; part <= 4; part += 2) {
                size_t pos = 0; int chars, len;
                while ((len = next_token(f[part], fl[part], pos, tok, chars)) > 0) {
                    uint32_t w = c.words.get_or_add(tok, len, isNew);
                    if (isNew) {
                        uint32_t key;
                        if (useStem) { int sl = stem::stem(tok, len, sb); key = c.keys.get_or_add(sb, sl, isNew); }
                        else key = c.keys.get_or_add(tok, len, isNew);
                        if (isNew) { c.keyWord.push(w); c.df.push(0); }
                        c.wordKey.push(key);
                    }
                    cur.push(c.wordKey[w]);
                }
            }
            qsort(cur.a, cur.n, 4, cmp_u32);
            size_t u = 0;
            for (size_t i = 0; i < cur.n; i++) if (!u || cur[u - 1] != cur[i]) cur[u++] = cur[i];
            DocTerms d; d.n = (uint32_t)u; d.t = (uint32_t *)malloc(u * 4 + 4);
            if (u) memcpy(d.t, cur.a, u * 4);
            for (size_t i = 0; i < u; i++) c.df[d.t[i]]++;
            c.docs.push(d);
        }
        cur.release();
    }
    fclose(in);
    printf("reference: %zu docs, %zu distinct keys, built in %.1f s\n", c.docs.n, c.keys.size, now_sec() - t0);

    Vec<uint32_t> order; order.resize(c.keys.size);
    for (uint32_t i = 0; i < c.keys.size; i++) order[i] = i;
    g_df = c.df.a;
    qsort(order.a, order.n, 4, cmp_df_desc);
    for (size_t i = 0; i < 300 && i < order.n; i++) c.frequent.push(order[i]);

    Index idx;
    if (!idx.load(indexPath)) return 1;
    EXPECT_EQ(idx.nDocs, c.docs.n);
    EXPECT_EQ(idx.stemmed, useStem);
    EXPECT_EQ(idx.nTerms, c.keys.size);
    int okCount = 0, bad = g_failed_checks;
    Vec<char> q; Vec<uint32_t> got, exp;
    double tq = 0;
    for (int i = 0; i < nq; i++) {
        q.clear(); c.nodes.clear();
        int root = gen_query(c, 0, q);
        exp.clear();
        for (uint32_t d = 0; d < c.docs.n; d++) if (eval_doc(c, root, c.docs[d])) exp.push(d);
        q.push(0);
        double ts = now_sec();
        run_query(idx, q.a, got);
        tq += now_sec() - ts;
        bool same = got.n == exp.n && (got.n == 0 || memcmp(got.a, exp.a, got.n * 4) == 0);
        g_checks++;
        if (same) okCount++;
        else {
            bad++; g_failed_checks++;
            if (bad <= 5) printf("    FAIL query \"%s\": index %zu docs, reference %zu\n", q.a, got.n, exp.n);
        }
    }
    printf("%d/%d random queries match the reference over all %zu documents (search time %.3f s total)\n",
           okCount, nq, c.docs.n, tq);
    printf(bad ? "FULL-CORPUS CHECK FAILED\n" : "FULL-CORPUS CHECK PASSED\n");
    return bad ? 1 : 0;
}

def log_sample(n, k=400):
    out, x = [], 1.0
    step = math.exp(math.log(n) / k)
    while x <= n:
        r = int(round(x))
        if not out or r != out[-1]:
            out.append(r)
        x *= step
    if out[-1] != n:
        out.append(n)
    return out


def linreg(xs, ys):
    mx, my = sum(xs) / len(xs), sum(ys) / len(ys)
    b = sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / sum((x - mx) ** 2 for x in xs)
    a = my - b * mx
    err = sum((y - a - b * x) ** 2 for x, y in zip(xs, ys)) / len(xs)
    return a, b, err


def mandelbrot(freqs):
    last = max(i for i, f in enumerate(freqs, 1) if f >= 3)
    ranks = log_sample(last)
    best = None
    for rho in [x / 2 for x in range(0, 401)]:
        xs = [math.log(r + rho) for r in ranks]
        ys = [math.log(freqs[r - 1]) for r in ranks]
        a, b, err = linreg(xs, ys)
        if best is None or err < best[3]:
            best = (math.exp(a), rho, -b, err)
    return best

static double linreg(const std::vector<double> &x, const std::vector<double> &y, double &a, double &b) {
    double n = (double)x.size(), sx = 0, sy = 0, sxx = 0, sxy = 0;
    for (size_t i = 0; i < x.size(); i++) { sx += x[i]; sy += y[i]; sxx += x[i] * x[i]; sxy += x[i] * y[i]; }
    b = (n * sxy - sx * sy) / (n * sxx - sx * sx);
    a = (sy - b * sx) / n;
    double e = 0;
    for (size_t i = 0; i < x.size(); i++) { double d = a + b * x[i] - y[i]; e += d * d; }
    return e / n;
}

static void mandelbrot(double &P, double &rho, double &B) {
    size_t last = 0;
    for (size_t i = 0; i < g_rows.size(); i++) if (g_rows[i].f >= 3) last = i + 1;
    std::vector<size_t> ranks;
    for (int k = 0; k < 400; k++) {
        size_t r = (size_t)std::lround(std::exp(std::log((double)last) * k / 399.0));
        if (ranks.empty() || ranks.back() != r) ranks.push_back(r);
    }
    double best = 1e300;
    for (int i = 0; i <= 400; i++) {

        double e = linreg(xs, ys, a, b);

def py_symbol(text, name):
    tree = ast.parse(text)
    parts = name.split('.')
    nodes = tree.body
    found = None
    for part in parts:
        found = next((n for n in nodes if isinstance(n, (ast.FunctionDef, ast.ClassDef, ast.AsyncFunctionDef))
                      and n.name == part), None)

    lines = text.splitlines()

def cpp_symbol(text, name):
    lines = text.splitlines()
    pat = re.compile(r'^\s*(?:template\s*<[^>]*>\s*)?(?:static\s+|inline\s+|const\s+)*'
                     r'(?:struct|class|enum|[\w:<>,\s\*&]+?)\s*\b' + re.escape(name) + r'\b\s*(?:\(|\{|:|$)'
                     r'|^\s*TEST\(\s*\w+\s*,\s*' + re.escape(name) + r'\s*\)')
    for i, l in enumerate(lines):
        if pat.search(l) and not l.strip().startswith('//') and not l.rstrip().endswith(';'):
            start = i
            while start > 0 and lines[start - 1].strip().startswith('//'):
                start -= 1
            depth, seen = 0, False
            for j in range(i, len(lines)):
                code = re.sub(r'"(?:\\.|[^"\\])*"|\'(?:\\.|[^\'\\])*\'|//.*', '', lines[j])
                depth += code.count('{') - code.count('}')
                seen = seen or '{' in code
                if seen and depth <= 0:
                    return '\n'.join(lines[start:j + 1]), start + 1
    return None
