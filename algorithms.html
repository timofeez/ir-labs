def collect_titles(name, max_cats):
    url, root, depth = SOURCES[name]
    seen_cat, pages = {root}, {}
    level = [root]
    for dep in range(depth + 1):
        nxt = []
        for cat in level:
            for m in members(url, cat):
                if m['ns'] == 0:
                    pages.setdefault(m['pageid'], m['title'])
                elif m['ns'] == 14 and m['title'] not in seen_cat and len(seen_cat) < max_cats:
                    seen_cat.add(m['title']); nxt.append(m['title'])
        print(f'{name}: depth {dep}, cats {len(seen_cat)}, pages {len(pages)}', flush=True)
        level = nxt
    return pages, len(seen_cat)

def clean_html(html):
    soup = BeautifulSoup(html, 'lxml')
    body = soup.select_one('article.article-block__main-body')
    header = soup.select_one('div.article-header')
    meta = {
        'issue': bool(header and header.find('h2') and 'Выпуск' in header.find('h2').get_text()),
        'photo': bool(header and header.select_one('.article-header__img img')),
        'subheadings': bool(body and body.find('b')),
        'author': bool(body and body.select_one('div.avt')),
        'rubric': False,
    }
    if body is None:
        return '', meta
    for el in body.select('div.avt, script, style'):
        el.decompose()
    for br in body.find_all('br'):
        br.replace_with('\n')
    text = body.get_text()
    m = RUBRIC.match(text)
    if m:
        meta['rubric'] = True
        text = text[m.end():]
    text = re.sub(r'[ \t\xa0]+', ' ', text)
    text = re.sub(r'\n\s*\n\s*(\n\s*)+', '\n\n', text)
    paragraphs = text.strip().split('\n\n')
    bold = body.find_all('b')
    last = paragraphs[-1].strip()
    if bold and bold[-1].get_text(strip=True) and bold[-1].get_text(strip=True) in last \
            and len(last.split()) <= 20 and len(paragraphs) > 1:
        meta['author'] = True
        paragraphs = paragraphs[:-1]
    return '\n\n'.join(p.strip() for p in paragraphs).strip(), meta

def clean(wt):
    t = TAIL.sub('', wt)
    t = COMMENT.sub('', t)
    t = REF.sub('', t)
    for _ in range(5):
        t2 = TEMPLATE.sub('', t)
        if t2 == t:
            break
        t = t2
    t = TABLE.sub('', t)
    t = FILE.sub('', t)
    t = LINK.sub(r'\1', t)
    t = EXTLINK.sub(r'\1', t)
    t = TAG.sub('', t)
    t = HEADING.sub(r'\1', t)
    t = FORMAT.sub('', t)

    t = re.sub(r'\n{3,}', '\n\n', t)
    return t.strip()

Нормализация URL#

Чтобы одна страница не скачивалась дважды под разными адресами: относительная ссылка → абсолютная, схема и хост в нижний регистр, удаление #фрагмента, служебных параметров, лишних /. Нормализованный URL — уникальный ключ в таблицах documents и frontier.

lab2/robot.py — normalize_url строки 42–66
def normalize_url(url, base=None):
    """Приводит URL к единому виду, чтобы одна страница не скачивалась дважды:
    схема и хост в нижнем регистре, http -> https для Википедии, мобильная версия -> обычная, без порта по умолчанию,
    без #фрагмента, единое %-кодирование пути, пробелы -> «_»,
    /w/index.php?title=X&pagefrom=Y -> /wiki/X?pagefrom=Y (путь /w/ запрещён в robots.txt Википедии),
    из параметров остаются только значимые (KEEP_QUERY), отсортированные."""
    if base:
        url = urljoin(base, url)
    p = urlsplit(url.strip())
    scheme = (p.scheme or 'https').lower()
    host = (p.hostname or '').lower()
    if scheme == 'http' and host.endswith('wikipedia.org'):
        scheme = 'https'                      # Википедия доступна только по https
    host = re.sub(r'^([a-z\-]+)\.m\.(wikipedia\.org)$', r'\1.\2', host)
    if p.port and p.port not in (80, 443):
        host = f'{host}:{p.port}'
    path = unquote(p.path).replace(' ', '_')
    query = [(k, v.replace(' ', '_')) for k, v in parse_qsl(p.query, keep_blank_values=False) if k in KEEP_QUERY]
    title = dict(query).get('title')
    query = [(k, v) for k, v in query if k != 'title']
    if path == '/w/index.php' and title:
        path = '/wiki/' + title
    path = quote(path, safe=SAFE)
    q = urlencode(sorted(query), quote_via=quote, safe=SAFE)
    return urlunsplit((scheme, host, path or '/', q, ''))

robots.txt и вежливость#

Для каждого хоста robots.txt скачивается один раз и кешируется; перед каждым запросом проверяется разрешение и выдерживается задержка между запросами к одному сайту.

lab2/robot.py — Robot.allowed строки 146–164
    def allowed(self, url):
        """Разрешает ли robots.txt сайта скачивать этот URL. robots.txt скачивается один раз на хост;
        если в нём указан Crawl-delay для нашего робота, задержка увеличивается до него."""
        if not self.respect_robots:
            return True
        p = urlsplit(url)
        key = f'{p.scheme}://{p.netloc}'
        if key not in self.robots:
            rp = urllib.robotparser.RobotFileParser()
            r = self.get(key + '/robots.txt')
            if r is not None and r.status_code == 200:
                rp.parse(r.text.splitlines())
            else:
                rp.allow_all = True                    # нет robots.txt — ограничений нет
            cd = rp.crawl_delay(self.http.headers['User-Agent'])
            if cd:
                self.delay = max(self.delay, float(cd))
            self.robots[key] = rp
        return self.robots[key].can_fetch(self.http.headers['User-Agent'], url)
lab2/robot.py — Robot.get строки 126–144
    def get(self, url, headers=None):
        """GET с паузой между запросами (logic.delay) и повторами при ошибках."""
        for attempt in range(self.retries):
            wait = self.delay - (time.perf_counter() - self.last_request)
            if wait > 0:
                time.sleep(wait)
            self.last_request = time.perf_counter()
            self.requests_done += 1
            try:
                r = self.http.get(url, headers=headers or {}, timeout=self.timeout)
                if r.status_code in (200, 304, 404):
                    return r
                if r.status_code == 429 or r.status_code >= 500:
                    time.sleep(min(60, 3 * 2 ** attempt))
                    continue
                return r
            except requests.RequestException:
                time.sleep(min(60, 3 * 2 ** attempt))
        return None

Обход очереди с возобновлением#

Очередь URL хранится в БД (frontier), а не в памяти. Робот берёт следующий необработанный URL (ORDER BY id — в порядке добавления, т.е. в ширину), скачивает, сохраняет документ (INSERT … ON CONFLICT DO NOTHING — повторная вставка не создаёт дубль) и помечает URL обработанным. После остановки и запуска обход продолжается с того же места.

    def crawl_frontier(self):
        """Обход очереди. Возвращает True, если что-то было сделано."""
        did = False
        while not self.stop:
            row = self.db.execute("SELECT id, url, source, kind, depth, is_cont, lastmod FROM frontier "
                                  "WHERE state='new' ORDER BY id LIMIT 1").fetchone()
            if not row:
                self.db.commit()          # закрыть читающую транзакцию: следующий шаг увидит свежие данные
                return did
            did = True
            fid, url, source, kind, depth, _, lastmod = row
            if kind == 'article' and self.db.execute('SELECT 1 FROM documents WHERE url=?', (url,)).fetchone():
                self.db.execute("UPDATE frontier SET state='done' WHERE id=?", (fid,))
                self.db.commit()
                continue
            if not self.allowed(url):
                self.db.execute("UPDATE frontier SET state='skipped', error='robots.txt' WHERE id=?", (fid,))
                self.db.commit()
                continue
            r = self.get(url)
            if r is None or r.status_code != 200:
                err = 'no response' if r is None else f'HTTP {r.status_code}'
                self.db.execute("UPDATE frontier SET state='error', error=? WHERE id=?", (err, fid))
            elif kind == 'category':
                self.process_category(row, r.text)
                self.db.execute("UPDATE frontier SET state='done' WHERE id=?", (fid,))
            else:
                dup = self.duplicate_of(url, source, r)
                if dup:                         # «Content seen?» — этот документ уже есть в базе
                    self.db.execute("UPDATE frontier SET state='duplicate', error=? WHERE id=?", (dup, fid))
                else:
                    self.save_document(self.canonical(url, r), source, r, int(time.time()), lastmod)
                    self.db.execute("UPDATE frontier SET state='done' WHERE id=?", (fid,))
            self.db.commit()   # после каждой страницы: остановка в любой момент ничего не теряет
            self.progress(kind, url)
            if self.max_requests and self.requests_done >= self.max_requests:
                self.stop = True

Повторный обход: обнаружение изменений#

Хеш (SHA-1) текста основного блока страницы сравнивается с сохранённым: если совпал — документ не перезаписывается; изменился — обновляется. Хешируется только содержательная часть, иначе меню или счётчики давали бы ложные «изменения».

lab2/robot.py — content_hash строки 80–87
def content_hash(html, selector):
    """SHA-1 содержимого статьи. Хешируется не вся страница, а только блок статьи (selector):
    меню, реклама и блоки «популярное» меняются от запроса к запросу, а статья — нет."""
    if selector:
        el = BeautifulSoup(html, 'lxml').select_one(selector)
        if el is not None:
            return hashlib.sha1(el.get_text(' ', strip=True).encode('utf-8')).hexdigest()
    return hashlib.sha1(html).hexdigest()

    def recrawl(self):
        """Переобкачка: проверяем документы, у которых истёк recrawl_interval.
        Источник с sitemap: карта сайта перечитывается, и страница запрашивается, только если её
        lastmod изменился. Остальные: условный GET If-Modified-Since. В обоих случаях новая версия
        сохраняется, только если изменилось содержимое статьи (хеш)."""
        did = False
        pass_start = int(time.time())
        due = pass_start - self.recrawl_interval
        fresh = {}                                    # источник -> {url: lastmod} из перечитанной карты сайта
        for s in self.sources.values():
            if s.get('type') == 'sitemap' and self.db.execute(
                    'SELECT 1 FROM documents WHERE source=? AND checked_at <= ? AND checked_at < ? LIMIT 1',
                    (s['name'], due, pass_start)).fetchone():
                fresh[s['name']] = self.read_sitemap(s)
        while not self.stop:
            now = int(time.time())
            # только документы, проверенные до начала этого прохода, — каждый проверяется один раз
            row = self.db.execute('SELECT url, source, last_modified, content_hash, hash_v, sitemap_lastmod '
                                  'FROM documents WHERE checked_at <= ? AND checked_at < ? ORDER BY checked_at LIMIT 1',
                                  (due, pass_start)).fetchone()
            if not row:
                self.db.commit()
                return did
            did = True
            url, source, last_modified, old_hash, hash_v, sm_old = row
            if source in fresh and fresh[source].get(url) == sm_old and sm_old:
                # дата в карте сайта та же — страница не менялась, запрос не нужен
                self.db.execute('UPDATE documents SET checked_at=? WHERE url=?', (now, url))
                status = 'sitemap: not modified'
            else:
                # условный запрос: сервер вернёт 304 Not Modified и пустое тело, если страница не менялась
                r = self.get(url, {'If-Modified-Since': last_modified} if last_modified else None)
                old = self.stored_hash(url, source, old_hash, hash_v)
                new_lastmod = fresh.get(source, {}).get(url, sm_old)
                if r is not None and r.status_code == 200 and content_hash(r.content, self.selector(source)) != old:
                    self.save_document(url, source, r, now, new_lastmod)   # изменилась — сохраняем новую версию
                    status = 'updated'
                else:
                    self.db.execute('UPDATE documents SET checked_at=?, sitemap_lastmod=? WHERE url=?',
                                    (now, new_lastmod, url))
                    status = 'not modified' if r is not None and r.status_code == 304 else 'same content'
            self.db.commit()
            self.progress('recrawl ' + status, url)
            if self.max_requests and self.requests_done >= self.max_requests:
                self.stop = True

def connect_any(hosts, database, user, autocommit=False):
    """Подключение к первому доступному узлу кластера."""
    last = None
    for h in hosts:
        host, _, port = h.partition(':')
        try:
            return psycopg.connect(host=host, port=int(port or 26257), user=user, dbname=database,
                                   autocommit=autocommit, connect_timeout=2,
                                   application_name='mai-ir-robot'), h
        except psycopg.OperationalError as e:
            last = e
    raise RuntimeError(f'ни один узел кластера не доступен ({", ".join(hosts)}): {last}')

  • Шардирование диапазонами ключа (range partitioning). Таблица упорядочена по первичному ключу и нарезана на диапазоны (ranges). Диапазон больше 512 МБ автоматически делится пополам. Декодирование UTF-8#

Количество байт символа определяется по старшим битам первого байта (0xxxxxxx — 1 байт, 110xxxxx — 2, 1110xxxx — 3, 11110xxx — 4); остальные байты дают по 6 бит. Битая последовательность заменяется и пропускается, чтобы токенизатор никогда не зацикливался.

// ---------------------------------------------------------------- UTF-8 и токенизация
// Декодирует один символ UTF-8 начиная с s[i], сдвигает i.
static inline uint32_t utf8_decode(const unsigned char *s, size_t n, size_t &i) {
    unsigned c = s[i];
    if (c < 0x80) { i++; return c; }
    int len = c >= 0xF0 ? 4 : c >= 0xE0 ? 3 : c >= 0xC0 ? 2 : 1;
    if (len == 1 || i + len > n) { i++; return 0xFFFD; }
    uint32_t cp = c & (0xFF >> (len + 1));
    for (int k = 1; k < len; k++) cp = (cp << 6) | (s[i + k] & 0x3F);
    i += len;
    return cp;
}

static inline int utf8_encode(uint32_t c, char *o) {
    if (c < 0x80) { o[0] = (char)c; return 1; }
    if (c < 0x800) { o[0] = (char)(0xC0 | (c >> 6)); o[1] = (char)(0x80 | (c & 0x3F)); return 2; }
    if (c < 0x10000) {
        o[0] = (char)(0xE0 | (c >> 12)); o[1] = (char)(0x80 | ((c >> 6) & 0x3F));
        o[2] = (char)(0x80 | (c & 0x3F)); return 3;
    }
    o[0] = (char)(0xF0 | (c >> 18)); o[1] = (char)(0x80 | ((c >> 12) & 0x3F));
    o[2] = (char)(0x80 | ((c >> 6) & 0x3F)); o[3] = (char)(0x80 | (c & 0x3F)); return 4;
}

Токенизация#

Конечный автомат по классам символов: буквы и цифры образуют токен; дефис и точка внутри слова («ВЛ-80», «3.5») оставляются, если с обеих сторон буквы/цифры; всё остальное — разделитель. Токен приводится к нижнему регистру, «ё» → «е», ударения удаляются. Один проход по тексту — O(n).

// Класс символа: 1 — часть слова, 2 — игнорируемый (знак ударения), 0 — разделитель.
static inline int char_class(uint32_t c) {
    if ((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9')) return 1;
    if (c >= 0x300 && c <= 0x36F) return 2;                 // комбинируемые диакритики (ударения)
    if (c >= 0xC0 && c <= 0x24F && c != 0xD7 && c != 0xF7) return 1;  // латиница с диакритикой
    if (c >= 0x370 && c <= 0x3FF) return 1;                 // греческий
    if (c >= 0x400 && c <= 0x4FF) return 1;                 // кириллица
    return 0;
}

    if (c >= 'A' && c <= 'Z') return c + 32;

    if (c >= 0xC0 && c <= 0xDE && c != 0xD7) return c + 0x20;

// Читает следующий токен из s[0..n) начиная с pos.
// В out пишется нормализованный терм (UTF-8, нижний регистр), возвращается его длина в байтах
// (0 — токенов больше нет). В chars — длина в символах.
static inline int next_token(const char *s, size_t n, size_t &pos, char *out, int &chars) {
    const unsigned char *u = (const unsigned char *)s;
    int len = 0; chars = 0;
    while (pos < n) {
        size_t start = pos;
        uint32_t c = utf8_decode(u, n, pos);
        int cls = char_class(c);
        if (cls == 1) {
            if (len + 4 <= MAX_TERM_BYTES) { len += utf8_encode(to_lower(c), out + len); chars++; }
        } else if (cls == 0) {
            if (len > 0) { pos = start; break; }
        }
        // cls == 2: знак ударения внутри слова просто пропускается
    }
    // разделитель, на котором остановились, пропустим при следующем вызове
    if (len > 0 && pos < n) { size_t p = pos; utf8_decode(u, n, p); pos = p; }
    return len;
}

ЛР4 — стемминг#

Русский: алгоритм Портера (Snowball)#

  1. Найти область RV — часть слова после первой гласной, и R2 — после второго сочетания «гласная + согласная».
  2. Шаг 1: отрезать окончание совершенного деепричастия; если нет — возвратную частицу (-ся/-сь), затем окончание прилагательного/причастия, глагола или существительного (всегда самое длинное из группы и только внутри RV).
  3. Шаг 2: отрезать «и». Шаг 3: словообразовательный суффикс «ост/ость» в R2.
  4. Шаг 4: «нн» → «н», суффикс превосходной степени «ейш(е)», мягкий знак.

Поиск самого длинного окончания — по таблицам окончаний (longest), отрезание — cut.

// Самое длинное окончание из таблицы на конце w[0..n), начинающееся не раньше lim. Возвращает его номер или -1.
static int longest(const Table &t, const uint32_t *w, int n, int lim) {
    int best = -1;
    for (int k = 0; k < t.n; k++) {
        const Suf &s = t.s[k];
        if (s.n > n - lim) continue;
        if (memcmp(w + n - s.n, s.c, s.n * sizeof(uint32_t)) != 0) continue;
        if (best < 0 || s.n > t.s[best].n) best = k;
    }
    return best;
}

// Как «among ... delete» в Snowball: берётся самое длинное окончание; если для него не выполнено

static bool cut(const Table &t, const uint32_t *w, int &n, int lim) {
    int k = longest(t, w, n, lim);
    if (k < 0) return false;
    int L = t.s[k].n;
    if (t.s[k].grp == 1) {
        int p = n - L - 1;
        if (p < lim || (w[p] != RU_A && w[p] != RU_YA)) return false;
    }
    n -= L;
    return true;
}

static int stem_ru(uint32_t *w, int n) {
    ru_init();
    // Области: RV — после первой гласной; R2 — область R1 внутри R1 (R1 — после первой согласной, идущей за гласной)
    int pV = n, p2 = n, i = 0;
    while (i < n && !ru_vowel(w[i])) i++;
    if (i < n) {
        pV = i + 1; i = pV;
        while (i < n && ru_vowel(w[i])) i++;          // до первой согласной
        if (i < n) {
            i++;                                       // i = p1
            while (i < n && !ru_vowel(w[i])) i++;
            if (i < n) {
                i++;
                while (i < n && ru_vowel(w[i])) i++;
                if (i < n) p2 = i + 1;
            }
        }
    }
    // Шаг 1: деепричастие, либо (возвратное «ся» + прилагательное/причастие, глагол или существительное)
    if (!cut(PERF, w, n, pV)) {
        cut(REFL, w, n, pV);
        int k = longest(ADJ, w, n, pV);
        if (k >= 0) {
            n -= ADJ.s[k].n;
            cut(PART, w, n, pV);                       // причастие перед окончанием прилагательного
        } else if (!cut(VERB, w, n, pV)) {
            cut(NOUN, w, n, pV);
        }
    }
    // Шаг 2: «и» на конце
    if (n - 1 >= pV && w[n - 1] == RU_I) n--;
    // Шаг 3: словообразовательный суффикс «ост(ь)» в R2
    int k = longest(DERIV, w, n, p2 > pV ? p2 : pV);
    if (k >= 0) n -= DERIV.s[k].n;
    // Шаг 4: превосходная степень «ейш(е)», двойное «нн» -> «н», мягкий знак
    k = longest(SUPER, w, n, pV);
    if (k >= 0) {
        n -= SUPER.s[k].n;
        if (n - 2 >= pV && w[n - 1] == RU_N && w[n - 2] == RU_N) n--;
    } else if (n - 2 >= pV && w[n - 1] == RU_N && w[n - 2] == RU_N) {
        n--;
    } else if (n - 1 >= pV && w[n - 1] == RU_SOFT) {
        n--;
    }
    return n;
}

    if (i < n) {
        i++; while (i < n && en_v(w[i])) i++;
        if (i < n) {
            p1 = i + 1; i = p1;
            while (i < n && !en_v(w[i])) i++;

        cut(REFL, w, n, pV);
        int k = longest(ADJ, w, n, pV);
        if (k >= 0) {
            n -= ADJ.s[k].n;
            cut(PART, w, n, pV);                       // причастие перед окончанием прилагательного
        } else if (!cut(VERB, w, n, pV)) {
            cut(NOUN, w, n, pV);
        }
    }
    // Шаг 2: «и» на конце
    if (n - 1 >= pV && w[n - 1] == RU_I) n--;
    // Шаг 3: словообразовательный суффикс «ост(ь)» в R2
    int k = longest(DERIV, w, n, p2 > pV ? p2 : pV);
    if (k >= 0) n -= DERIV.s[k].n;
    // Шаг 4: превосходная степень «ейш(е)», двойное «нн» -> «н», мягкий знак
    k = longest(SUPER, w, n, pV);
    if (k >= 0) {
        n -= SUPER.s[k].n;
        if (n - 2 >= pV && w[n - 1] == RU_N && w[n - 2] == RU_N) n--;
    } else if (n - 2 >= pV && w[n - 1] == RU_N && w[n - 2] == RU_N) {
        n--;
    } else if (n - 1 >= pV && w[n - 1] == RU_SOFT) {
        n--;
    }
    return n;
}

static bool en_v(char c) { return c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' || c == 'y'; }

static bool ends(const char *w, int n, const char *s) {
    int L = (int)strlen(s);
    return L <= n && memcmp(w + n - L, s, L) == 0;
}
// «Короткий слог» на конце w[0..n): согласная (не w, x, Y), перед ней гласная, перед ней согласная
static bool shortv(const char *w, int n) {
    if (n < 3) return false;
    char c = w[n - 1];
    return !en_v(c) && c != 'w' && c != 'x' && c != 'Y' && en_v(w[n - 2]) && !en_v(w[n - 3]);
}
static bool has_vowel(const char *w, int n) {
    for (int i = 0; i < n; i++) if (en_v(w[i])) return true;
    return false;
}

struct Rule { const char *suf; const char *rep; };

// Самое длинное правило из списка, подходящее к концу слова; -1, если ни одно не подходит
static int en_longest(const Rule *r, int cnt, const char *w, int n) {
    int best = -1, bl = -1;
    for (int k = 0; k < cnt; k++) {
        int L = (int)strlen(r[k].suf);
        if (L > bl && ends(w, n, r[k].suf)) { best = k; bl = L; }
    }
    return best;
}
static void replace_end(char *w, int &n, int cut_len, const char *rep) {
    n -= cut_len;
    int L = (int)strlen(rep);
    memcpy(w + n, rep, L);
    n += L;
}

    if (n <= 0) return n;
    // «y» — согласная в начале слова и после гласной (помечается как 'Y')
    if (w[0] == 'y') w[0] = 'Y';
    for (int i = 1; i < n; i++) if (w[i] == 'y' && en_v(w[i - 1])) w[i] = 'Y';
    // Области R1, R2
    int p1 = n, p2 = n, i = 0;
    while (i < n && !en_v(w[i])) i++;
    if (i < n) {
        i++; while (i < n && en_v(w[i])) i++;
        if (i < n) {
            p1 = i + 1; i = p1;
            while (i < n && !en_v(w[i])) i++;
            if (i < n) { i++; while (i < n && en_v(w[i])) i++; if (i < n) p2 = i + 1; }
        }
    }

    if (ends(w, n, "sses")) n -= 2;
    else if (ends(w, n, "ies")) n -= 2;
    else if (ends(w, n, "ss")) {}
    else if (ends(w, n, "s")) n -= 1;
    // Шаг 1b

    } else {

        }

    // Шаг 4 (в R2)
    static const Rule S4[] = {
        {"al", ""}, {"ance", ""}, {"ence", ""}, {"er", ""}, {"ic", ""}, {"able", ""}, {"ible", ""},
        {"ant", ""}, {"ement", ""}, {"ment", ""}, {"ent", ""}, {"ou", ""}, {"ism", ""}, {"ate", ""},
        {"iti", ""}, {"ous", ""}, {"ive", ""}, {"ize", ""}, {"ion", ""}};

    if (k >= 0) {
        int L = (int)strlen(S4[k].suf);
        if (n - L >= p2) {

    }

        if (c >= p2 || (c >= p1 && !shortv(w, c))) n--;
    }
    // Шаг 5b: «ll» -> «l» в R2
    if (n >= 2 && w[n - 1] == 'l' && n - 1 >= p2 && w[n - 2] == 'l') n--;
    for (int j = 0; j < n; j++) if (w[j] == 'Y') w[j] = 'y';
    return n;
}

// ---------------------------------------------------------------- общий вход
// Основа терма s[0..n) (UTF-8, нижний регистр) -> out; возвращает длину основы в байтах.
static int stem(const char *s, int n, char *out) {
    bool en = n > 0, ru = n > 0;
    uint32_t cp[MAXW]; int m = 0;
    size_t i = 0;
    while (i < (size_t)n && m < MAXW) {
        uint32_t c = utf8_decode((const unsigned char *)s, n, i);
        cp[m++] = c;
        if (!(c >= 'a' && c <= 'z')) en = false;
        if (!(c >= 0x430 && c <= 0x44F)) ru = false;
    }
    if (en) {
        memcpy(out, s, n);
        return stem_en(out, n);
    }
    if (ru) {
        m = stem_ru(cp, m);
        int len = 0;
        for (int j = 0; j < m; j++) len += utf8_encode(cp[j], out + len);
        return len;
    }
    memcpy(out, s, n);
    return n;
}

// в порядке первого появления терма в корпусе).
static void sort_by_freq(uint32_t *a, size_t n) {
    uint32_t *tmp = (uint32_t *)malloc(n * sizeof(uint32_t));
    uint32_t *src = a, *dst = tmp;
    for (size_t width = 1; width < n; width *= 2) {
        for (size_t lo = 0; lo < n; lo += 2 * width) {
            size_t mid = lo + width < n ? lo + width : n, hi = lo + 2 * width < n ? lo + 2 * width : n;
            size_t i = lo, j = mid, k = lo;
            while (i < mid && j < hi) dst[k++] = g_freq[src[j]] > g_freq[src[i]] ? src[j++] : src[i++];
            while (i < mid) dst[k++] = src[i++];
            while (j < hi) dst[k++] = src[j++];

        uint32_t *t = src; src = dst; dst = t;
    }
    if (src != a) memcpy(a, src, n * sizeof(uint32_t));

def linreg(xs, ys):
    mx, my = sum(xs) / len(xs), sum(ys) / len(ys)
    b = sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / sum((x - mx) ** 2 for x in xs)
    a = my - b * mx
    err = sum((y - a - b * x) ** 2 for x, y in zip(xs, ys)) / len(xs)
    return a, b, err

            if (type == SEC_DOCS) docs = p;
            else if (type == SEC_DICT) dict = p;
            else if (type == SEC_TERMSTR) termstr = p;
            else if (type == SEC_POSTINGS) postings = p;
            // неизвестные секции пропускаются — так формат расширяется без поломки старых программ
        }
        if (!docs || !dict || !termstr || !postings) { fprintf(stderr, "missing section\n"); return false; }
        nDocs = rd32(docs);
        nTerms = rd32(dict);
        stemmed = (rd32(data + 12) & FLAG_STEMMED) != 0;   // флаги заголовка
        return true;
    }

Лексер и парсер (рекурсивный спуск)#

Грамматика (приоритет: ! выше И, И выше ИЛИ):

or    := and ( '||' and )*
and   := unary ( ['&&'] unary )*      пробел между словами = И
unary := '!' unary | '(' or ')' | TERM

Каждому правилу соответствует функция; результат — дерево запроса в массиве узлов. Парсер толерантен: лишние операторы пропускаются, незакрытые скобки закрываются.

static void lex(const Index &idx, const char *q, size_t n, Vec<QTok> &out) {
    out.clear();
    size_t i = 0;
    char tok[MAX_TERM_BYTES + 4], sb[MAX_TERM_BYTES + 8];
    while (i < n) {
        char c = q[i];
        if (c == ' ' || c == '\t' || c == '\r' || c == '\n') { i++; continue; }
        if (c == '(') { out.push({T_LP, 0}); i++; continue; }
        if (c == ')') { out.push({T_RP, 0}); i++; continue; }
        if (c == '!') { out.push({T_NOT, 0}); i++; continue; }
        if (c == '&') { while (i < n && q[i] == '&') i++; out.push({T_AND, 0}); continue; }
        if (c == '|') { while (i < n && q[i] == '|') i++; out.push({T_OR, 0}); continue; }
        // слово: всё до пробела или спецсимвола; внутри него работает тот же токенизатор, что и в индексаторе
        size_t st = i;
        while (i < n && !strchr(" \t\r\n()!&|", q[i])) i++;
        size_t pos = 0; int chars, len;
        bool first = true;
        while ((len = next_token(q + st, i - st, pos, tok, chars)) > 0) {
            // «ВЛ-80» даёт два терма — между ними неявное И
            if (!first) out.push({T_AND, 0});
            if (idx.stemmed) {                     // тот же стеммер, что при индексации
                int sl = stem::stem(tok, len, sb);
                out.push({T_TERM, idx.find_term(sb, sl)});
            } else {
                out.push({T_TERM, idx.find_term(tok, len)});
            }
            first = false;
        }
    }
}

struct Parser {
    const Vec<QTok> &t;
    size_t p = 0;
    Vec<Node> nodes;
    explicit Parser(const Vec<QTok> &toks) : t(toks) {}

    int make(NodeType ty, int l, int r, long e = 0) { nodes.push({ty, l, r, e}); return (int)nodes.n - 1; }
    bool at(TokType ty) const { return p < t.n && t[p].type == ty; }

    int unary() {
        if (at(T_NOT)) { p++; int c = unary(); return c < 0 ? -1 : make(N_NOT, c, -1); }
        if (at(T_LP)) { p++; int e = expr(); if (at(T_RP)) p++; return e; }
        if (at(T_TERM)) { long e = t[p].entry; p++; return make(N_TERM, -1, -1, e); }
        return -1;
    }
    int conj() {
        int left = unary();
        for (;;) {
            if (at(T_AND)) { p++; continue; }               // «&&» и повторные «&&» просто разделители
            if (at(T_TERM) || at(T_NOT) || at(T_LP)) {
                int right = unary();
                if (right < 0) continue;
                left = left < 0 ? right : make(N_AND, left, right);
                continue;
            }
            return left;
        }
    }
    int expr() {
        int left = conj();
        while (at(T_OR)) {
            p++;
            int right = conj();
            if (right < 0) continue;
            left = left < 0 ? right : make(N_OR, left, right);
        }
        return left;
    }
    int parse() {
        int root = expr();
        while (p < t.n) {                                   // хвост после лишней «)» — склеиваем через И
            p++;
            int more = expr();
            if (more >= 0) root = root < 0 ? more : make(N_AND, root, more);
        }
        return root;
    }
};

Операции над отсортированными списками#

Два указателя идут по спискам одновременно, на каждом шаге продвигается тот, у кого номер меньше. Пересечение (И), объединение (ИЛИ), разность (И НЕ) — O(|A| + |B|). Отрицание отдельного терма — дополнение до всех документов, O(N). Запрос a !b вычисляется как разность без построения дополнения.

static void op_and(const Vec<uint32_t> &a, const Vec<uint32_t> &b, Vec<uint32_t> &r) {
    r.clear(); size_t i = 0, j = 0;
    while (i < a.n && j < b.n) {
        if (a[i] < b[j]) i++;
        else if (a[i] > b[j]) j++;
        else { r.push(a[i]); i++; j++; }
    }
}

static void op_or(const Vec<uint32_t> &a, const Vec<uint32_t> &b, Vec<uint32_t> &r) {
    r.clear(); r.reserve(a.n + b.n); size_t i = 0, j = 0;
    while (i < a.n && j < b.n) {
        if (a[i] < b[j]) r.push(a[i++]);
        else if (a[i] > b[j]) r.push(b[j++]);
        else { r.push(a[i]); i++; j++; }
    }
    while (i < a.n) r.push(a[i++]);
    while (j < b.n) r.push(b[j++]);
}

// ---------------------------------------------------------------- вычисление
static void eval(const Index &idx, const Vec<Node> &nodes, int v, Vec<uint32_t> &res) {
    const Node &nd = nodes[v];
    if (nd.type == N_TERM) { idx.postings_of(nd.entry, res); return; }
    Vec<uint32_t> a, b;
    if (nd.type == N_NOT) { eval(idx, nodes, nd.l, a); op_not(a, idx.nDocs, res); a.release(); return; }
    if (nd.type == N_AND && nodes[nd.r].type == N_NOT) {    // A && !B — вычитание без построения дополнения
        eval(idx, nodes, nd.l, a); eval(idx, nodes, nodes[nd.r].l, b); op_and_not(a, b, res);
    } else if (nd.type == N_AND && nodes[nd.l].type == N_NOT) {
        eval(idx, nodes, nd.r, a); eval(idx, nodes, nodes[nd.l].l, b); op_and_not(a, b, res);
    } else {
        eval(idx, nodes, nd.l, a); eval(idx, nodes, nd.r, b);
        if (nd.type == N_AND) op_and(a, b, res); else op_or(a, b, res);
    }
    a.release(); b.release();
}
